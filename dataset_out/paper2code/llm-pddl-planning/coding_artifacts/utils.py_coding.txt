def save_pddl_files(domain_code: str, problem_code: str, out_dir: str) -> None:
    """Write domain and problem PDDL code to files in specified directory."""
    os.makedirs(out_dir, exist_ok=True)
    domain_path = os.path.join(out_dir, 'domain.pddl')
    problem_path = os.path.join(out_dir, 'problem.pddl')
    try:
        with open(domain_path, 'w', encoding='utf-8') as f_dom:
            f_dom.write(domain_code)
        with open(problem_path, 'w', encoding='utf-8') as f_prob:
            f_prob.write(problem_code)
        logger.info(f"PDDL files saved: {domain_path}, {problem_path}")
    except Exception as e:
        logger.exception(f"Failed to save PDDL files: {e}")

def load_pddl_files(domain_path: str, problem_path: str) -> Tuple[str, str]:
    """Load PDDL strings from files."""
    try:
        with open(domain_path, 'r', encoding='utf-8') as f:
            domain_code = f.read()
        with open(problem_path, 'r', encoding='utf-8') as f:
            problem_code = f.read()
        return domain_code, problem_code
    except Exception as e:
        logger.exception(f"Error loading PDDL files: {e}")
        return "", ""

def run_external_planner(domain_file: str, problem_file: str, timeout: int = 300) -> Optional[List[str]]:
    """Run classical planner (e.g., FastDownward) and return plan actions."""
    plan_file = 'plan.txt'  # or a temp file as needed
    cmd = [
        PLANNER_PATH,
        '--search', 'astar(lmcut())',
        '--plan-file', plan_file,
        domain_file,
        problem_file
    ]
    try:
        start_time = time.time()
        result = subprocess.run(cmd, capture_output=True, text=True, timeout=timeout)
        dur = time.time() - start_time
        if result.returncode != 0:
            logger.info(f"Planner exited with code {result.returncode}")
            return None
        # Parse plan file
        if not os.path.exists(plan_file):
            return None
        actions = []
        with open(plan_file, 'r', encoding='utf-8') as f:
            for line in f:
                line_strip = line.strip()
                if line_strip and not line_strip.startswith(';'):
                    # Plan lines typically are action lines
                    actions.append(line_strip)
        return actions if actions else None
    except subprocess.TimeoutExpired:
        logger.warning(f"Planning timed out after {timeout} seconds.")
        return None
    except Exception as e:
        logger.exception(f"Error running planner: {e}")
        return None

def plan_exists(domain_file: str, problem_file: str, timeout: int = 300) -> bool:
    """Check if a plan exists without returning the plan."""
    plan = run_external_planner(domain_file, problem_file, timeout)
    return plan is not None

def execute_action(env_api: Dict, environment, action_str: str) -> bool:
    """Execute the action in the environment via API."""
    try:
        url = env_api.get('execute_url', f"{API_BASE_URL}/execute")
        payload = {'action': action_str}
        resp = requests.post(url, json=payload)
        resp.raise_for_status()
        res_json = resp.json()
        success = res_json.get('status', '').lower() == 'success'
        if success:
            # Optionally update environment state if provided
            environment.current_state = resp_json.get('state', environment.current_state)
        return success
    except Exception as e:
        logger.warning(f"Action execution failed: {e}")
        return False

def check_executability(env_api: Dict, environment, action_sequence: List[str]) -> bool:
    """Check if a sequence of actions is executable."""
    try:
        url = env_api.get('check_url', f"{API_BASE_URL}/check")
        payload = {'actions': action_sequence}
        resp = requests.post(url, json=payload)
        resp.raise_for_status()
        res_json = resp.json()
        return res_json.get('executable', False)
    except Exception as e:
        logger.warning(f"Executability check failed: {e}")
        return False

def validate_plan(env_api: Dict, environment, plan: List[str]) -> bool:
    """Validate plan in environment (V_{d,p})."""
    try:
        url = env_api.get('validate_url', f"{API_BASE_URL}/validate")
        payload = {'plan': plan}
        resp = requests.post(url, json=payload)
        resp.raise_for_status()
        res_json = resp.json()
        return res_json.get('valid', False)
    except Exception as e:
        logger.warning(f"Plan validation failed: {e}")
        return False

            # Filter executable actions

            if not executable_actions:
                break
            chosen_act = random.choice(executable_actions)
            success = execute_action(env_api, environment, chosen_act)
            if not success:
                success_flag = False
                break
            seq.append(chosen_act)
        # Check full sequence executability
        if success_flag and check_executability(env_api, environment, seq):
            sampled_sequences.append(seq)
        # Restore environment state if needed
        environment.current_state = env_state_backup
    return sampled_sequences

def generate_possible_actions(environment, env_api: Dict) -> List[str]:
    """Generate list of candidate actions from environment's action interface."""
    # Placeholder: assuming environment.get_action_interface() provides actions
    return list(environment.get_action_interface().get('actions', []))

def compute_ew_score(environment, env_api: Dict, domain_str: str, problem_str: str, T_max: int) -> float:
    """Compute EW score based on sampled action sequences."""
    sequences = sample_random_action_sequence(environment, env_api, domain_str, problem_str, T_max)
    if not sequences:
        return 0.0
    exec_count = sum(1 for seq in sequences if check_executability(env_api, environment, seq))
    return exec_count / len(sequences)

def run_plan_and_validate(environment, env_api: Dict, domain_str: str, problem_str: str) -> bool:
    """Run classical planner, then validate plan in environment."""
    # Save PDDL files temporarily
    save_pddl_files(domain_str, problem_str, 'temp_pddl')
    domain_fp = os.path.abspath(os.path.join('temp_pddl', 'domain.pddl'))
    problem_fp = os.path.abspath(os.path.join('temp_pddl', 'problem.pddl'))
    plan = run_external_planner(domain_fp, problem_fp)
    if plan is None:
        return False
    # Validate plan in environment
    return validate_plan(env_api, environment, plan)

def get_current_time() -> float:

def log_message(msg: str, level: str = 'info') -> None:
    """Log message with level."""
    if level == 'info':
        logger.info(msg)
    elif level == 'warning':
        logger.warning(msg)
    elif level == 'debug':
        logger.debug(msg)
    elif level == 'error':
        logger.error(msg)

def compute_collection_stats(success_list: List[int], ew_scores: List[float]) -> Dict:
    """Compute overall stats, success rate, average EW, etc."""
    total = len(success_list)
    success_count = sum(success_list)
    success_rate = success_count / total if total > 0 else 0.0
    avg_ew = np.mean(ew_scores) if ew_scores else 0.0
    return {
        'success_rate': success_rate,
        'average_ew': avg_ew,
        'total': total,
        'success_count': success_count
    }
