## pddl_utils.py

import os
from typing import Tuple, List, Dict, Union, Optional
import subprocess
import tempfile
import logging

try:
    import pddlpy
except ImportError:
    # If pddlpy is not installed, raise informative error later
    pddlpy = None

class PDDLUtils:
    """
    Utility class for handling PDDL syntax validation, parsing, predicate extraction,
    and file management, supporting the pipeline with essential correctness checks.
    """
    def __init__(self):
        """
        Initialize the PDDLUtils instance.
        """
        # Setting up logger for debug outputs
        self.logger = logging.getLogger(__name__)
        self.logger.setLevel(logging.INFO)

    def validate_pddl_syntax(self, pddl_code: str) -> bool:
        """
        Validates PDDL syntax by attempting to parse the code using pddlpy.
        Args:
            pddl_code (str): PDDL code string.
        Returns:
            bool: True if syntax is valid, False otherwise.
        """
        if pddlpy is None:
            self.logger.warning("pddlpy library not installed. Cannot validate syntax.")
            return False
        try:
            # Write PDDL code temporarily to parse
            with tempfile.NamedTemporaryFile(mode='w', suffix='.pddl', delete=False, encoding='utf-8') as temp_file:
                temp_file.write(pddl_code)
                temp_filename = temp_file.name
            # Parse domain
            parser = pddlpy.Parser(domain=temp_filename)
            # For the syntax validation, just parsing the domain suffices
            # attempt to parse the domain; if no exception, syntax is considered valid
            os.unlink(temp_filename)
            return True
        except Exception as e:
            self.logger.debug(f"Syntax validation failed: {e}")
            return False

    def parse_pddl(self, pddl_code: str) -> Optional[Dict[str, Union[Dict, List]]]:
        """
        Parses the PDDL code into structured data using pddlpy.
        Args:
            pddl_code (str): PDDL code string.
        Returns:
            dict or None: Parsed representation with 'domain' and 'problem' keys, or None if parsing fails.
        """
        if pddlpy is None:
            self.logger.warning("pddlpy library not installed. Cannot parse PDDL.")
            return None
        try:
            # Write to temporary file
            with tempfile.NamedTemporaryFile(mode='w', suffix='.pddl', delete=False, encoding='utf-8') as temp_file:
                temp_file.write(pddl_code)
                filename = temp_file.name
            # Determine if code is domain or problem
            # Here, we attempt parsing both
            parser_domain = pddlpy.Parser(domain=filename)
            # For this basic implementation, return the parser object
            # For detailed access, extract predicates, actions, etc.
            parsed_data = {
                'domain': parser_domain.domain,
                # 'problem': parser_problem # optional if needed
            }
            os.unlink(filename)
            return parsed_data
        except Exception as e:
            self.logger.debug(f"Parsing failed: {e}")
            return None

    def extract_predicates(self, pddl_code: str) -> List[Dict]:
        """
        Extracts predicates from a parsed domain PDDL.
        Args:
            pddl_code (str): PDDL domain string.
        Returns:
            List[Dict]: List of predicate descriptors with name and argument list.
        """
        if pddlpy is None:
            self.logger.warning("pddlpy library not installed. Cannot extract predicates.")
            return []

        try:
            # Write to temporary file
            with tempfile.NamedTemporaryFile(mode='w', suffix='.pddl', delete=False, encoding='utf-8') as temp_file:
                temp_file.write(pddl_code)
                filename = temp_file.name
            parser = pddlpy.Parser(domain=filename)
            predicates = []
            if parser.domain and parser.domain.predicates:
                for pred in parser.domain.predicates:
                    pred_dict = {
                        'name': pred.name,
                        'arguments': [(arg.name, arg.type) for arg in pred.args]
                    }
                    predicates.append(pred_dict)
            os.unlink(filename)
            return predicates
        except Exception as e:
            self.logger.debug(f"Predicate extraction failed: {e}")
            return []

    def write_pddl_files(self, domain_str: str, problem_str: str, out_dir: str) -> None:
        """
        Writes the domain and problem PDDL strings to files in the specified directory.
        Args:
            domain_str (str): PDDL code for domain.
            problem_str (str): PDDL code for problem.
            out_dir (str): Directory path to write files.
        """
        try:
            os.makedirs(out_dir, exist_ok=True)
            domain_path = os.path.join(out_dir, 'domain.pddl')
            problem_path = os.path.join(out_dir, 'problem.pddl')
            with open(domain_path, 'w', encoding='utf-8') as f_domain:
                f_domain.write(domain_str)
            with open(problem_path, 'w', encoding='utf-8') as f_problem:
                f_problem.write(problem_str)
            self.logger.info(f"PDDL files written: {domain_path}, {problem_path}")
            # Optional: validate syntax after writing
            if not self.validate_pddl_syntax(domain_str):
                self.logger.warning("Generated domain PDDL may have syntax errors.")
            if not self.validate_pddl_syntax(problem_str):
                self.logger.warning("Generated problem PDDL may have syntax errors.")
        except Exception as e:
            self.logger.exception(f"Error writing PDDL files: {e}")

    def validate_all(self, domain_str: str, problem_str: str) -> bool:
        """
        Validates both domain and problem syntax.
        Args:
            domain_str (str): Domain PDDL code.
            problem_str (str): Problem PDDL code.
        Returns:
            bool: True if both valid, False otherwise.
        """
        valid_domain = self.validate_pddl_syntax(domain_str)
        valid_problem = self.validate_pddl_syntax(problem_str)
        return valid_domain and valid_problem

    def extract_predicates_from_domain(self, domain_str: str) -> List[Dict]:
        """
        Convenience method to extract predicates from domain string.
        Args:
            domain_str (str): Domain PDDL code.
        Returns:
            List[Dict]: List of predicate info.
        """
        return self.extract_predicates(domain_str)

    def read_pddl_file(self, filepath: str) -> str:
        """
        Reads PDDL code from a given file.
        Args:
            filepath (str): Path to PDDL file.
        Returns:
            str: PDDL code content.
        """
        with open(filepath, 'r', encoding='utf-8') as f:
            return f.read()

    def parse_pddl_file(self, filepath: str) -> Optional[Dict]:
        """
        Parses a PDDL file into structured data.
        Args:
            filepath (str): Path to PDDL file.
        Returns:
            dict or None: Parsed representation or None if error.
        """
        with open(filepath, 'r', encoding='utf-8') as f:
            pddl_code = f.read()
        return self.parse_pddl(pddl_code)
