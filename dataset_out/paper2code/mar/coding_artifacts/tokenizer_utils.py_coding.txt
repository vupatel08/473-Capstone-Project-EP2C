# --- Main class ---
class Tokenizer:
    """
    Provides a unified interface for different tokenizer types: VQ or KL.
    """
    def __init__(self, tokenizer_type: str = 'VQ-16', tokenizer_path: str = '', device: str = 'cuda'):
        assert tokenizer_type in ['VQ-16', 'KL-16'], f"Unsupported tokenizer_type: {tokenizer_type}"
        self.tokenizer_type = tokenizer_type
        self.tokenizer_path = tokenizer_path
        self.device = device

        if tokenizer_type.startswith('VQ'):
            # Load VQ-GAN based tokenizer
            self.encoder, self.decoder, self.codebook = load_vqgan(tokenizer_path)
            self.encoder.to(device).eval()
            self.decoder.to(device).eval()
            self.codebook.embeddings.to(device)
            self.mode = 'discrete'
        elif tokenizer_type.startswith('KL'):
            # Load KL-VAE based tokenizer
            self.encoder, self.decoder = load_kl(tokenizer_path)
            self.encoder.to(device).eval()
            self.decoder.to(device).eval()
            self.mode = 'continuous'
        else:
            raise ValueError(f"Unknown tokenizer type: {tokenizer_type}")

    def encode(self, image):
        """
        Encodes an input PIL image or tensor into tokens.
        Returns:
            For VQ: np.ndarray of shape [sequence_length], dtype=int
            For KL: torch.Tensor of shape [sequence_length, feature_dim]
        """
        # Convert image to tensor if PIL
        if isinstance(image, Image.Image):
            img_tensor = TF.to_tensor(image).unsqueeze(0).to(self.device)
        elif isinstance(image, torch.Tensor):
            img_tensor = image.unsqueeze(0).to(self.device)
        else:
            raise TypeError("Input must be PIL Image or torch.Tensor")
        # Resize image to expected size if needed
        # Assuming preprocessed outside as 256x256

        with torch.no_grad():
            if self.mode == 'discrete':
                # VQ-GAN: encode and get codebook indices
                feats = self.encoder(img_tensor)  # shape: [1,C,H,W]
                C, H, W = feats.shape[1], feats.shape[2], feats.shape[3]
                # Quantize features to codebook indices
                # Compute similarity with codebook embeddings
                # For simplicity, assume features are already quantized
                # but typically perform nearest neighbor search
                flattened = feats.permute(0,2,3,1).reshape(-1, C)  # [H*W, C]
                codebook_matrix = self.codebook.embeddings.weight  # [K, C]
                # Compute distances
                dists = (
                    torch.sum(flattened**2, dim=1, keepdim=True)
                    - 2 * torch.matmul(flattened, codebook_matrix.t())
                    + torch.sum(codebook_matrix**2, dim=1)
                )  # [H*W, K]
                indices = torch.argmin(dists, dim=1)  # [H*W]
                tokens = indices.reshape(H, W).cpu().numpy().astype(np.int32)
                return tokens.flatten()  # [sequence_length]
            elif self.mode == 'continuous':
                # KL: encode to latent features
                latents = self.encoder(img_tensor)  # [1, feat_dim, H, W]
                # Normalize if needed
                feats = latents.squeeze(0).permute(1,2,0)  # [H, W, feat_dim]
                return feats.reshape(-1, feats.shape[-1])  # [sequence_length, feat_dim]
            else:
                raise ValueError(f"Unknown mode: {self.mode}")

    def decode(self, tokens):
        """
        Decodes tokens into a PIL Image.
        Args:
            tokens: list/array of indices (for VQ) or tensor of features (for KL)
        Returns:
            PIL Image
        """
        with torch.no_grad():
            if self.mode == 'discrete':
                # Map indices to codebook embeddings
                indices = torch.tensor(tokens).to(self.device).long()
                embeddings = self.codebook.lookup(indices.unsqueeze(0))  # shape: [1, seq_len, embed_dim]
                # Reshape to feature map
                seq_len = embeddings.shape[1]
                size = int(np.sqrt(seq_len))
                feat_map = embeddings.reshape(1, size, size, -1).permute(0,3,1,2)  # [1,C,H,W]
                # Decode to image
                recon_img = self.decoder(feat_map)  # [1,3,H,W]
                # Convert to PIL
                img_tensor = torch.clamp(recon_img.squeeze(0).cpu(), 0, 1)
                img = TF.to_pil_image(img_tensor)
                return img
            elif self.mode == 'continuous':
                # Use continuous features to generate image
                feats = torch.tensor(tokens).to(self.device)  # shape: [sequence_length, feature_dim]
                size = int(np.sqrt(feats.shape[0]))
                feats = feats.reshape(size, size, -1).permute(2,0,1).unsqueeze(0)  # [1,C,H,W]
                # Decode to image
                recon_img = self.decoder(feats)  # [1,3,H,W]
                img_tensor = torch.clamp(recon_img.squeeze(0).cpu(), 0, 1)
                img = TF.to_pil_image(img_tensor)
                return img
            else:
                raise ValueError(f"Unknown mode: {self.mode}")

# --- Singleton cache for loaded tokenizer ---
_tokenizer_cache = {}

def get_tokenizer(tokenizer_type: str = 'VQ-16', tokenizer_path: str = '', device: str = 'cuda'):
    global _tokenizer_cache
    key = f"{tokenizer_type}_{tokenizer_path}"
    if key in _tokenizer_cache:
        return _tokenizer_cache[key]
    tokenizer = Tokenizer(tokenizer_type, tokenizer_path, device=device)
    _tokenizer_cache[key] = tokenizer
    return tokenizer
