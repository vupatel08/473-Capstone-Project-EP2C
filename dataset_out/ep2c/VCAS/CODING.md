## dataset_loader.py
import random
from typing import Optional, Dict, Any, List, Tuple, Union
import torch
from torch.utils.data import Dataset, DataLoader
from datasets import load_dataset
from transformers import AutoTokenizer
from torchvision import datasets as vision_datasets
from torchvision import transforms

class DatasetLoader:
    """
    DatasetLoader handles dataset loading, tokenization, preprocessing,
    batching, and shuffling for various tasks (NLP and vision).
    """
    def __init__(self,
                 dataset_name: str = 'glue',
                 task_name: str = 'sst2',
                 split: str = 'train',
                 tokenizer_name: str = 'bert-base-uncased',
                 max_tokens: int = 128,
                 batch_size: int = 32,
                 shuffle: bool = True,
                 seed: int = 42,
                 max_image_size: int = 224,
                 image_transform: Optional[transforms.Compose] = None):
        """
        Initializes DatasetLoader with dataset configuration.

        Args:
            dataset_name (str): Name of the dataset (e.g., 'glue', 'imagenet', etc.)
            task_name (str): Specific task/dataset subset (e.g., 'sst2', 'mnli', 'cifar10')
            split (str): Dataset split ('train', 'validation', 'test')
            tokenizer_name (str): Name of the transformer tokenizer to use
            max_tokens (int): Max tokens per sample for NLP
            batch_size (int): Batch size for DataLoader
            shuffle (bool): Whether to shuffle dataset each epoch
            seed (int): Random seed for reproducibility
            max_image_size (int): Max image size for vision datasets
            image_transform (Callable): Optional torchvision transforms for images
        """
        self.dataset_name = dataset_name
        self.task_name = task_name
        self.split = split
        self.tokenizer_name = tokenizer_name
        self.max_tokens = max_tokens
        self.batch_size = batch_size
        self.shuffle = shuffle
        self.seed = seed
        self.max_image_size = max_image_size
        self.image_transform = image_transform
        self.dataset = None
        self.tokenizer = None

        self._load_tokenizer()
        self._load_dataset()

    def _load_tokenizer(self):
        """Initialize the tokenizer from transformers."""
        self.tokenizer = AutoTokenizer.from_pretrained(self.tokenizer_name)

    def _load_dataset(self):
        """Load dataset based on dataset_name and task_name."""
        # We differentiate between NLP (GLUE) and vision datasets
        if self.dataset_name.lower() == 'glue':
            # For GLUE, load specific task
            raw_ds = load_dataset('glue', self.task_name)
            dataset_split = raw_ds[self.split]
            # Convert to list for easy processing
            self.dataset = dataset_split
        elif self.dataset_name.lower() in ['imagenet', 'cifar10', 'cifar100', 'cifar100']:
            # For vision datasets, load via torchvision
            if self.dataset_name.lower() == 'imagenet':
                # Placeholder: requires ImageNet dataset path
                # For simplicity, assume data in standard folder 'imagenet_root/train'
                # User may need to modify path accordingly
                # Here, we do NOT implement full loading of ImageNet due to complexity
                raise NotImplementedError("ImageNet loading requires dataset path.")
            elif self.dataset_name.lower() == 'cifar10':
                self.dataset = vision_datasets.CIFAR10(root='./data', train=(self.split=='train'),
                                                        transform=self._get_image_transform(),
                                                        download=True)
            elif self.dataset_name.lower() in ['cifar100']:
                self.dataset = vision_datasets.CIFAR100(root='./data', train=(self.split=='train'),
                                                         transform=self._get_image_transform(),
                                                         download=True)
            else:
                raise ValueError(f"Unsupported vision dataset: {self.dataset_name}")
        else:
            raise ValueError(f"Unsupported dataset: {self.dataset_name}")

    def _get_image_transform(self):
        """Return torchvision transforms for images."""
        if self.image_transform is not None:
            return self.image_transform
        # Default transform: resize + normalize
        return transforms.Compose([
            transforms.Resize((self.max_image_size, self.max_image_size)),
            transforms.ToTensor(),
            transforms.Normalize(mean=[0.485, 0.456, 0.406],
                                 std=[0.229, 0.224, 0.225]),
        ])

    def get_loader(self) -> DataLoader:
        """
        Returns DataLoader for the dataset.
        For NLP datasets: custom collate_fn with tokenization.
        For vision datasets: default DataLoader handles tensor returns.
        """
        if self.dataset_name.lower() == 'glue':
            # For NLP, wrap Dataset with tokenization in collate_fn
            return DataLoader(self._get_nlp_dataset(),
                              batch_size=self.batch_size,
                              shuffle=self.shuffle,
                              collate_fn=self._nlp_collate_fn,
                              drop_last=False,
                              pin_memory=True)
        elif self.dataset_name.lower() in ['imagenet', 'cifar10', 'cifar100', 'cifar100']:
            # For vision, direct DataLoader
            return DataLoader(self.dataset,
                              batch_size=self.batch_size,
                              shuffle=self.shuffle,
                              drop_last=False,
                              pin_memory=True)
        else:
            raise ValueError("Dataset not loaded properly.")

    def _get_nlp_dataset(self):
        """Wrap raw dataset to a PyTorch Dataset with tokenization."""
        class NLPPipelineDataset(Dataset):
            def __init__(self, data, tokenizer, max_tokens):
                self.data = data
                self.tokenizer = tokenizer
                self.max_tokens = max_tokens

            def __len__(self):
                return len(self.data)

            def __getitem__(self, idx):
                item = self.data[idx]
                # For SST-2: label is under 'label', text under 'sentence' or 'question' keys
                # Adjust depending on dataset keys
                if 'sentence' in item:
                    text = item['sentence']
                elif 'question' in item:
                    text = item['question']
                elif 'sentence1' in item and 'sentence2' in item:
                    # For QQP etc.
                    text = (item['sentence1'], item['sentence2'])
                else:
                    raise KeyError("Unrecognized data format for NLP dataset.")
                label = item['label']
                # Tokenize with padding/truncation
                encoding = self.tokenizer(
                    text,
                    max_length=self.max_tokens,
                    padding='max_length',
                    truncation=True,
                    return_tensors='pt'
                )
                # flatten tensors
                input_ids = encoding['input_ids'].squeeze(0)
                attention_mask = encoding['attention_mask'].squeeze(0)
                return {
                    'input_ids': input_ids,
                    'attention_mask': attention_mask,
                    'labels': torch.tensor(label, dtype=torch.long)
                }
        return NLPPipelineDataset(self.dataset, self.tokenizer, self.max_tokens)

    def _nlp_collate_fn(self, batch: List[Dict[str, torch.Tensor]]) -> Dict[str, torch.Tensor]:
        """
        Collate function to combine batch tensors for NLP.
        Args:
            batch: list of dicts with 'input_ids', 'attention_mask', 'labels'
        Returns:
            Dict with batch tensors stacked
        """
        input_ids = torch.stack([item['input_ids'] for item in batch])
        attention_mask = torch.stack([item['attention_mask'] for item in batch])
        labels = torch.stack([item['labels'] for item in batch])
        return {
            'input_ids': input_ids,
            'attention_mask': attention_mask,
            'labels': labels
        }

## evaluation.py
import torch
from torch.utils.data import DataLoader
from datasets import load_dataset
from transformers import AutoTokenizer, AutoModelForSequenceClassification
from sklearn.metrics import accuracy_score, f1_score
from typing import Dict, List, Optional
import os

class Evaluation:
    """
    Loads a trained model and dataset, runs inference, and computes specified evaluation metrics.
    """

    def __init__(self, config: Dict):
        """
        Initializes evaluation with configuration, loads model and dataset.
        Args:
            config (Dict): configuration loaded from 'config.yaml'
        """
        # Extract model config
        model_type: str = config['model'].get('type', 'bert-base-uncased')
        pretrained: bool = config['model'].get('pretrained', True)
        max_seq_length: int = config['model'].get('max_seq_length', 128)
        # Dataset config
        dataset_name: str = config['dataset'].get('name', 'glue')
        task_name: str = config['dataset'].get('task', 'sst2')
        split: str = 'validation' if 'validation' in config['dataset'].get('split', '') else 'test'
        tokenizer_name: str = config['dataset'].get('tokenizer_name', 'bert-base-uncased')

        # Load tokenizer
        self.tokenizer = AutoTokenizer.from_pretrained(tokenizer_name)

        # Load dataset
        self.dataset = self._load_dataset(dataset_name, task_name, split)

        # DataLoader
        batch_size = 32  # set small batch for evaluation
        self.dataloader = DataLoader(self.dataset, batch_size=batch_size, shuffle=False, pin_memory=True)

        # Load model
        self.device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
        self.model = self._load_model(model_type, pretrained)
        self.model.to(self.device)
        self.model.eval()

        # Collect metrics to compute
        self.metrics = config['evaluation'].get('metrics', ['accuracy', 'loss'])

        # Loss criterion
        self.criterion = torch.nn.CrossEntropyLoss()

    def _load_dataset(self, dataset_name: str, task_name: str, split: str):
        """
        Loads the dataset from datasets library
        """
        if dataset_name.lower() == 'glue':
            raw_ds = load_dataset('glue', task_name)
            # Choose split
            if split not in raw_ds:
                raise ValueError(f"Split '{split}' not found in dataset.")
            dataset_split = raw_ds[split]
        else:
            raise ValueError(f"Unsupported dataset: {dataset_name}")
        return dataset_split

    def _load_model(self, model_type: str, pretrained: bool):
        """
        Loads the model architecture and weights from checkpoint if available.
        """
        # Load model: assume model weights are saved under a standardized path
        # For this code, replace 'checkpoint.pth' with actual path if needed
        model_path = 'checkpoint.pth'  # adjust as needed
        model = AutoModelForSequenceClassification.from_pretrained(model_type)
        if os.path.exists(model_path):
            state_dict = torch.load(model_path, map_location='cpu')
            model.load_state_dict(state_dict)
        else:
            # If no checkpoint, load default pretrained
            if pretrained:
                model = AutoModelForSequenceClassification.from_pretrained(model_type)
        return model

    def evaluate(self) -> Dict[str, float]:
        """
        Runs inference over the dataset, computes metrics, returns dict.
        """
        total_loss = 0.0
        total_samples = 0
        all_predictions = []
        all_labels = []

        with torch.no_grad():
            for batch in self.dataloader:
                inputs = self._prepare_batch(batch).to(self.device)
                labels = batch['labels'].to(self.device)

                outputs = self.model(**inputs)
                logits = outputs.logits

                # Compute loss if needed
                if 'loss' in self.metrics:
                    loss = self.criterion(logits, labels)
                    total_loss += loss.item() * labels.size(0)
                else:
                    loss = None

                preds = torch.argmax(logits, dim=-1)
                all_predictions.extend(preds.cpu().numpy())
                all_labels.extend(labels.cpu().numpy())

                total_samples += labels.size(0)

        # Compute metrics
        results = {}
        if 'accuracy' in self.metrics:
            acc = accuracy_score(all_labels, all_predictions)
            results['accuracy'] = acc
        if 'loss' in self.metrics:
            results['loss'] = total_loss / total_samples
        # Add other metrics if needed
        if 'f1' in self.metrics:
            f1 = f1_score(all_labels, all_predictions, average='macro')
            results['f1'] = f1

        return results

    def _prepare_batch(self, batch: Dict) -> Dict:
        """
        Tokenizes data batch, returns model inputs dict.
        """
        # For datasets like SST-2, batch contains text data. For others, adjust accordingly.
        # Assuming batch contains 'sentence' or similar text key.
        if 'sentence' in self.dataset.features:
            texts = batch['sentence']
            encoding = self.tokenizer(
                texts,
                max_length=self.max_seq_length,
                padding='max_length',
                truncation=True,
                return_tensors='pt'
            )
        elif 'sentence1' in self.dataset.features or 'sentence2' in self.dataset.features:
            # e.g., QNLI, MRPC
            texts = list(zip(batch['sentence1'], batch['sentence2']))
            # For simplicity, concatenate sentence pairs
            texts = ['{} [SEP] {}'.format(s1, s2) for s1, s2 in texts]
            encoding = self.tokenizer(
                texts,
                max_length=self.max_seq_length,
                padding='max_length',
                truncation=True,
                return_tensors='pt'
            )
        else:
            # fallback: assume input_ids present
            # Possibly for other tasks
            return {
                'input_ids': batch['input_ids'],
                'attention_mask': batch['attention_mask']
            }
        return {
            'input_ids': encoding['input_ids'],
            'attention_mask': encoding['attention_mask']
        }

## importance_sampler.py
import torch
from typing import List, Dict, Optional

class ImportanceSampler:
    """
    Implements importance ratio management and importance-driven sampling
    for data points (activation gradients) and tokens (weight gradients),
    supporting their adaptive update based on variance estimates.
    """
    def __init__(
        self,
        layer_num: int,
        initial_ratio: float = 1.0,
        importance_method: str = 'gradient_norm',
        activation_threshold: float = 0.025,
        weight_threshold: float = 0.025,
        update_step_alpha: float = 0.01,
        ratio_scaling_beta: float = 0.95,
        variance_update_frequency: int = 100
    ):
        """
        Args:
            layer_num (int): Number of layers in the model.
            initial_ratio (float): Initial importance ratio for all layers.
            importance_method (str): Method to compute importance ('gradient_norm' currently supported).
            activation_threshold (float): Variance threshold for activation importance ratio (tau_act).
            weight_threshold (float): Variance threshold for weight importance ratio (tau_w).
            update_step_alpha (float): Step size for updating importance ratios.
            ratio_scaling_beta (float): Multiplier for importance ratios during update.
            variance_update_frequency (int): How often (in steps) to update importance ratios.
        """
        self.layer_num = layer_num
        self.importance_method = importance_method
        # Importance ratios for activation gradients per layer
        self.rho: List[float] = [initial_ratio for _ in range(layer_num)]
        # Importance ratios for token weights per layer
        self.nu: List[float] = [initial_ratio for _ in range(layer_num)]
        # Variance control thresholds
        self.tau_act = activation_threshold
        self.tau_w = weight_threshold
        # Hyperparameters for adaptive update
        self.alpha = update_step_alpha
        self.beta = ratio_scaling_beta
        self.F = variance_update_frequency
        # Keep track of training step
        self.step = 0

        # Optionally, store historical importance scores or variance for more sophisticated schemes
        # But for simplicity, we rely on current variance estimates provided externally.

    def update_ratios(self, variance_estimates: Dict[str, List[float]]) -> None:
        """
        Updates importance ratios (rho and nu) based on estimated variances.
        Args:
            variance_estimates (dict): Dictionary with keys 'activation' and 'weight', 
            each mapping to a list of per-layer variance estimates.
            Example:
            {
                'activation': [var_layer_1, ..., var_layer_L],
                'weight': [var_layer_1, ..., var_layer_L]
            }
        """
        # Variance estimates are assumed to be provided as lists for each layer
        # Each list's elements correspond to layer-wise variance estimates
        activation_vars = variance_estimates.get('activation', [])
        weight_vars = variance_estimates.get('weight', [])

        for l in range(self.layer_num):
            # Update activation importance ratio rho_l
            if len(activation_vars) > l:
                var_act = activation_vars[l]
                # Target is self.tau_act
                # Adjust rho_l to control variance, e.g. proportional to ratio of variance to threshold
                ratio_act = self.rho[l]
                # Increase ratio if variance is small, decrease if too large
                # Clamp between 0 and 1
                new_rho = max(0.0, min(1.0, ratio_act * (var_act / self.tau_act)))
                # Alternatively, use exponential smoothing:
                self.rho[l] += self.alpha * (new_rho - self.rho[l])
                self.rho[l] = max(0.0, min(1.0, self.rho[l]))
            # Update weight importance ratio nu_l
            if len(weight_vars) > l:
                var_w = weight_vars[l]
                new_nu = max(0.0, min(1.0, self.nu[l] * (var_w / self.tau_w)))
                self.nu[l] += self.alpha * (new_nu - self.nu[l])
                self.nu[l] = max(0.0, min(1.0, self.nu[l]))

    def sample_indices(self, importance_scores: torch.Tensor, ratio: float) -> torch.Tensor:
        """
        Sample data point indices based on importance scores.
        Args:
            importance_scores (Tensor): 1D tensor with importance scores per data point (size N).
            ratio (float): Current importance ratio for data points (0 < ratio <=1).

        Returns:
            mask (Tensor): Boolean tensor of shape (N,), True for sampled points.
        """
        N = importance_scores.shape[0]
        # Convert importance scores to probabilities proportional to importance_scores
        # Ensure sum of probabilities equals ratio * N
        # To avoid small probabilities being zero, add epsilon
        epsilon = 1e-8
        importance_scores = importance_scores + epsilon
        total = importance_scores.sum()
        # Scale probabilities so that sum equals ratio * N
        probs = importance_scores * (ratio * N / total)
        probs = torch.clamp(probs, max=1.0)  # Probabilities capped at 1

        # Bernoulli sampling for each data point
        sampled_mask = torch.bernoulli(probs)
        # Make sure the expected number of samples is close to ratio * N
        return sampled_mask.bool()

    def sample_tokens(self, token_importance_scores: torch.Tensor, ratio: float) -> torch.Tensor:
        """
        Sample tokens within sequences based on importance scores.
        Args:
            token_importance_scores (Tensor): shape [N, T], importance scores of each token.
            ratio (float): importance ratio for tokens in [0,1].

        Returns:
            mask (Tensor): Boolean tensor of shape [N, T], True for retained tokens.
        """
        N, T = token_importance_scores.shape
        # Flatten scores to shape [N*T]
        scores_flat = token_importance_scores.reshape(-1) + 1e-8
        total_score = scores_flat.sum()
        # Total number of tokens across batch
        total_tokens = N * T
        # Compute probabilities proportional to importance scores
        probs = scores_flat * (ratio * total_tokens / total_score)
        probs = torch.clamp(probs, max=1.0)

        # Bernoulli sampling for each token
        sampled_mask_flat = torch.bernoulli(probs)
        sampled_mask = sampled_mask_flat.reshape(N, T).bool()
        return sampled_mask

    def get_activation_importance_scores(self, layer_activation_grads: List[torch.Tensor]) -> List[torch.Tensor]:
        """
        Compute importance scores for activation gradients in each layer.
        Currently: based on gradient norm.
        Args:
            layer_activation_grads (List[Tensor]): List of activation gradients per layer.
            Each tensor shape: [N, T, K] or similar.

        Returns:
            List of importance scores (e.g., Frobenius norms per sample per layer).
        """
        importance_scores: List[torch.Tensor] = []
        for grad in layer_activation_grads:
            # Compute per-sample importance: Frobenius norm over K for each sample
            # Assuming shape [N, T, K]
            # Compute norm over last dimension for each [N, T]
            norm_per_sample = torch.norm(grad, dim=-1, p='fro')  # shape [N, T]
            # Sum over tokens to get per-sample importance
            importance = norm_per_sample.sum(dim=1)  # shape [N]
            importance_scores.append(importance)
        return importance_scores

    def get_weight_importance_scores(self, layer_weight_grads: List[torch.Tensor]) -> List[torch.Tensor]:
        """
        Compute importance scores for weight gradients (e.g., leverage scores).
        For simplicity, use Frobenius norms.
        Args:
            layer_weight_grads (List[Tensor]): List of weight gradient tensors per layer.

        Returns:
            List of importance scores (scalar per layer).
        """
        importance_scores: List[torch.Tensor] = []
        for grad in layer_weight_grads:
            importance = torch.norm(grad, p='fro')
            importance_scores.append(importance)
        return importance_scores

    def update_step(self):
        """
        Increment step count and trigger ratios update if needed.
        """
        self.step += 1
        if self.step % self.F == 0:

## main.py
import yaml
import torch
import random
import numpy as np

from dataset_loader import DatasetLoader
from model import get_model
from importance_sampler import ImportanceSampler
from variance_estimator import VarianceEstimator
from trainer import Trainer
from evaluation import Evaluation

def main():
    # Load configuration from 'config.yaml'
    with open('config.yaml', 'r') as f:
        config = yaml.safe_load(f)

    # Set random seeds for reproducibility
    seed = 42
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed)

    # Device setup
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')

    # ==========================
    # 1. Dataset Loading
    # ==========================
    dataset_cfg = config.get('dataset', {})
    dataset_loader = DatasetLoader(**dataset_cfg)
    train_loader = dataset_loader.get_loader()

    # Optionally, load validation set for evaluation
    val_loader = None
    if 'split' in dataset_cfg:
        # For eval, use validation split if available
        val_cfg = dataset_cfg.copy()
        val_cfg['split'] = 'validation' if 'validation' in dataset_cfg['split'] else 'test'
        val_loader_obj = DatasetLoader(**val_cfg)
        val_loader = val_loader_obj.get_loader()

    # ============================
    # 2. Model Initialization
    # ============================
    model_cfg = config.get('model', {})
    model = get_model(model_cfg)
    model.to(device)

    # ============================
    # 3. Importance Sampler & Variance Estimator
    # ============================
    layer_num = None
    # Infer number of layers from model
    # For simplicity, assume model has attribute 'num_layers' or infer from model modules.
    if hasattr(model, 'model'):
        for name, module in model.model.named_modules():
            if isinstance(module, torch.nn.Linear):
                layer_num = getattr(model, 'num_layers', None)
                break
    if layer_num is None:
        # Fallback: assume model has attribute or set default
        layer_num = 12  # default, or derive dynamically

    importance_method = config.get('importance_sampling', {}).get('importance_method', 'gradient_norm')
    init_ratio = config.get('importance_sampling', {}).get('initial_ratio', 1.0)

    importance_sampler = ImportanceSampler(
        layer_num=layer_num,
        initial_ratio=init_ratio,
        importance_method=importance_method,
        activation_threshold=config['hyperparameters'].get('variance_control_thresholds', {}).get('activation', 0.025),
        weight_threshold=config['hyperparameters'].get('variance_control_thresholds', {}).get('weight', 0.025),
        update_step_alpha=config['hyperparameters'].get('update_step_alpha', 0.01),
        ratio_scaling_beta=config['hyperparameters'].get('ratio_scaling_beta', 0.95),
        variance_update_frequency=config['hyperparameters'].get('variance_update_frequency', 100)
    )

    M = config['hyperparameters'].get('monte_carlo_samples', 4)
    variance_estimator = VarianceEstimator(num_samples=M)

    # ============================
    # 4. Hyperparameters & Training Setup
    # ============================
    total_steps = config['training'].get('total_steps', 10000)
    epochs = config['training'].get('epochs', 3)
    batch_size = config['training'].get('batch_size', 32)
    lr = config['training'].get('learning_rate', 2e-5)
    warmup_steps = config['training'].get('warmup_steps', 500)

    update_ratio_freq = config['hyperparameters'].get('variance_update_frequency', 100)
    tau_act = config['hyperparameters'].get('variance_control_thresholds', {}).get('activation', 0.025)
    tau_w = config['hyperparameters'].get('variance_control_thresholds', {}).get('weight', 0.025)
    alpha = config['hyperparameters'].get('update_step_alpha', 0.01)
    beta = config['hyperparameters'].get('ratio_scaling_beta', 0.95)

    # Initialize sample ratios for each layer (activation and token)
    importance_sampler.rho = [1.0 for _ in range(layer_num)]
    importance_sampler.nu = [1.0 for _ in range(layer_num)]

    # ==============================
    # 5. Instantiate Trainer
    # ==============================
    trainer = Trainer(
        model=model,
        dataset_config=dataset_cfg,
        importance_sampler=importance_sampler,
        variance_estimator=variance_estimator,
        hyperparameters={
            'variance_control_thresholds': {'activation': tau_act, 'weight': tau_w},
            'update_step_alpha': alpha,
            'ratio_scaling_beta': beta,
            'variance_update_frequency': update_ratio_freq,
            'monte_carlo_samples': M
        },
        train_loader=train_loader,
        total_steps=total_steps,
        num_epochs=epochs,
        learning_rate=lr,
        warmup_steps=warmup_steps,
        device=device
    )

    # =======================
    # 6. Training & Variance Adaptation
    # =======================
    print("Starting training with VCAS...")
    trainer.train()

    # ============================
    # 7. Final Evaluation
    # ============================
    print("Training completed. Running evaluation...")
    eval_obj = Evaluation(model, dataset_loader)  # Use dataset_loader for validation/test
    results = eval_obj.evaluate()

    # Print evaluation results

    """

    in transformers, and can be adapted for other linear layers.
