## dataset_loader.py
import random
from typing import Optional, Dict, Any, List, Tuple, Union
import torch
from torch.utils.data import Dataset, DataLoader
from datasets import load_dataset
from transformers import AutoTokenizer
from torchvision import datasets as vision_datasets
from torchvision import transforms

class DatasetLoader:
    """
    DatasetLoader handles dataset loading, tokenization, preprocessing,
    batching, and shuffling for various tasks (NLP and vision).
    """
    def __init__(self,
                 dataset_name: str = 'glue',
                 task_name: str = 'sst2',
                 split: str = 'train',
                 tokenizer_name: str = 'bert-base-uncased',
                 max_tokens: int = 128,
                 batch_size: int = 32,
                 shuffle: bool = True,
                 seed: int = 42,
                 max_image_size: int = 224,
                 image_transform: Optional[transforms.Compose] = None):
        """
        Initializes DatasetLoader with dataset configuration.

        Args:
            dataset_name (str): Name of the dataset (e.g., 'glue', 'imagenet', etc.)
            task_name (str): Specific task/dataset subset (e.g., 'sst2', 'mnli', 'cifar10')
            split (str): Dataset split ('train', 'validation', 'test')
            tokenizer_name (str): Name of the transformer tokenizer to use
            max_tokens (int): Max tokens per sample for NLP
            batch_size (int): Batch size for DataLoader
            shuffle (bool): Whether to shuffle dataset each epoch
            seed (int): Random seed for reproducibility
            max_image_size (int): Max image size for vision datasets
            image_transform (Callable): Optional torchvision transforms for images
        """
        self.dataset_name = dataset_name
        self.task_name = task_name
        self.split = split
        self.tokenizer_name = tokenizer_name
        self.max_tokens = max_tokens
        self.batch_size = batch_size
        self.shuffle = shuffle
        self.seed = seed
        self.max_image_size = max_image_size
        self.image_transform = image_transform
        self.dataset = None
        self.tokenizer = None

        self._load_tokenizer()
        self._load_dataset()

    def _load_tokenizer(self):
        """Initialize the tokenizer from transformers."""
        self.tokenizer = AutoTokenizer.from_pretrained(self.tokenizer_name)

    def _load_dataset(self):
        """Load dataset based on dataset_name and task_name."""
        # We differentiate between NLP (GLUE) and vision datasets
        if self.dataset_name.lower() == 'glue':
            # For GLUE, load specific task
            raw_ds = load_dataset('glue', self.task_name)
            dataset_split = raw_ds[self.split]
            # Convert to list for easy processing
            self.dataset = dataset_split
        elif self.dataset_name.lower() in ['imagenet', 'cifar10', 'cifar100', 'cifar100']:
            # For vision datasets, load via torchvision
            if self.dataset_name.lower() == 'imagenet':
                # Placeholder: requires ImageNet dataset path
                # For simplicity, assume data in standard folder 'imagenet_root/train'
                # User may need to modify path accordingly
                # Here, we do NOT implement full loading of ImageNet due to complexity
                raise NotImplementedError("ImageNet loading requires dataset path.")
            elif self.dataset_name.lower() == 'cifar10':
                self.dataset = vision_datasets.CIFAR10(root='./data', train=(self.split=='train'),
                                                        transform=self._get_image_transform(),
                                                        download=True)
            elif self.dataset_name.lower() in ['cifar100']:
                self.dataset = vision_datasets.CIFAR100(root='./data', train=(self.split=='train'),
                                                         transform=self._get_image_transform(),
                                                         download=True)
            else:
                raise ValueError(f"Unsupported vision dataset: {self.dataset_name}")
        else:
            raise ValueError(f"Unsupported dataset: {self.dataset_name}")

    def _get_image_transform(self):
        """Return torchvision transforms for images."""
        if self.image_transform is not None:
            return self.image_transform
        # Default transform: resize + normalize
        return transforms.Compose([
            transforms.Resize((self.max_image_size, self.max_image_size)),
            transforms.ToTensor(),
            transforms.Normalize(mean=[0.485, 0.456, 0.406],
                                 std=[0.229, 0.224, 0.225]),
        ])

    def get_loader(self) -> DataLoader:
        """
        Returns DataLoader for the dataset.
        For NLP datasets: custom collate_fn with tokenization.
        For vision datasets: default DataLoader handles tensor returns.
        """
        if self.dataset_name.lower() == 'glue':
            # For NLP, wrap Dataset with tokenization in collate_fn
            return DataLoader(self._get_nlp_dataset(),
                              batch_size=self.batch_size,
                              shuffle=self.shuffle,
                              collate_fn=self._nlp_collate_fn,
                              drop_last=False,
                              pin_memory=True)
        elif self.dataset_name.lower() in ['imagenet', 'cifar10', 'cifar100', 'cifar100']:
            # For vision, direct DataLoader
            return DataLoader(self.dataset,
                              batch_size=self.batch_size,
                              shuffle=self.shuffle,
                              drop_last=False,
                              pin_memory=True)
        else:
            raise ValueError("Dataset not loaded properly.")

    def _get_nlp_dataset(self):
        """Wrap raw dataset to a PyTorch Dataset with tokenization."""
        class NLPPipelineDataset(Dataset):
            def __init__(self, data, tokenizer, max_tokens):
                self.data = data
                self.tokenizer = tokenizer
                self.max_tokens = max_tokens

            def __len__(self):
                return len(self.data)

            def __getitem__(self, idx):
                item = self.data[idx]
                # For SST-2: label is under 'label', text under 'sentence' or 'question' keys
                # Adjust depending on dataset keys
                if 'sentence' in item:
                    text = item['sentence']
                elif 'question' in item:
                    text = item['question']
                elif 'sentence1' in item and 'sentence2' in item:
                    # For QQP etc.
                    text = (item['sentence1'], item['sentence2'])
                else:
                    raise KeyError("Unrecognized data format for NLP dataset.")
                label = item['label']
                # Tokenize with padding/truncation
                encoding = self.tokenizer(
                    text,
                    max_length=self.max_tokens,
                    padding='max_length',
                    truncation=True,
                    return_tensors='pt'
                )
                # flatten tensors
                input_ids = encoding['input_ids'].squeeze(0)
                attention_mask = encoding['attention_mask'].squeeze(0)
                return {
                    'input_ids': input_ids,
                    'attention_mask': attention_mask,
                    'labels': torch.tensor(label, dtype=torch.long)
                }
        return NLPPipelineDataset(self.dataset, self.tokenizer, self.max_tokens)

    def _nlp_collate_fn(self, batch: List[Dict[str, torch.Tensor]]) -> Dict[str, torch.Tensor]:
        """
        Collate function to combine batch tensors for NLP.
        Args:
            batch: list of dicts with 'input_ids', 'attention_mask', 'labels'
        Returns:
            Dict with batch tensors stacked
        """
        input_ids = torch.stack([item['input_ids'] for item in batch])
        attention_mask = torch.stack([item['attention_mask'] for item in batch])
        labels = torch.stack([item['labels'] for item in batch])
        return {
            'input_ids': input_ids,
            'attention_mask': attention_mask,
            'labels': labels
        }
