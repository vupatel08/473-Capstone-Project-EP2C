## importance_sampler.py
import torch
from typing import List, Dict, Optional

class ImportanceSampler:
    """
    Implements importance ratio management and importance-driven sampling
    for data points (activation gradients) and tokens (weight gradients),
    supporting their adaptive update based on variance estimates.
    """
    def __init__(
        self,
        layer_num: int,
        initial_ratio: float = 1.0,
        importance_method: str = 'gradient_norm',
        activation_threshold: float = 0.025,
        weight_threshold: float = 0.025,
        update_step_alpha: float = 0.01,
        ratio_scaling_beta: float = 0.95,
        variance_update_frequency: int = 100
    ):
        """
        Args:
            layer_num (int): Number of layers in the model.
            initial_ratio (float): Initial importance ratio for all layers.
            importance_method (str): Method to compute importance ('gradient_norm' currently supported).
            activation_threshold (float): Variance threshold for activation importance ratio (tau_act).
            weight_threshold (float): Variance threshold for weight importance ratio (tau_w).
            update_step_alpha (float): Step size for updating importance ratios.
            ratio_scaling_beta (float): Multiplier for importance ratios during update.
            variance_update_frequency (int): How often (in steps) to update importance ratios.
        """
        self.layer_num = layer_num
        self.importance_method = importance_method
        # Importance ratios for activation gradients per layer
        self.rho: List[float] = [initial_ratio for _ in range(layer_num)]
        # Importance ratios for token weights per layer
        self.nu: List[float] = [initial_ratio for _ in range(layer_num)]
        # Variance control thresholds
        self.tau_act = activation_threshold
        self.tau_w = weight_threshold
        # Hyperparameters for adaptive update
        self.alpha = update_step_alpha
        self.beta = ratio_scaling_beta
        self.F = variance_update_frequency
        # Keep track of training step
        self.step = 0

        # Optionally, store historical importance scores or variance for more sophisticated schemes
        # But for simplicity, we rely on current variance estimates provided externally.

    def update_ratios(self, variance_estimates: Dict[str, List[float]]) -> None:
        """
        Updates importance ratios (rho and nu) based on estimated variances.
        Args:
            variance_estimates (dict): Dictionary with keys 'activation' and 'weight', 
            each mapping to a list of per-layer variance estimates.
            Example:
            {
                'activation': [var_layer_1, ..., var_layer_L],
                'weight': [var_layer_1, ..., var_layer_L]
            }
        """
        # Variance estimates are assumed to be provided as lists for each layer
        # Each list's elements correspond to layer-wise variance estimates
        activation_vars = variance_estimates.get('activation', [])
        weight_vars = variance_estimates.get('weight', [])

        for l in range(self.layer_num):
            # Update activation importance ratio rho_l
            if len(activation_vars) > l:
                var_act = activation_vars[l]
                # Target is self.tau_act
                # Adjust rho_l to control variance, e.g. proportional to ratio of variance to threshold
                ratio_act = self.rho[l]
                # Increase ratio if variance is small, decrease if too large
                # Clamp between 0 and 1
                new_rho = max(0.0, min(1.0, ratio_act * (var_act / self.tau_act)))
                # Alternatively, use exponential smoothing:
                self.rho[l] += self.alpha * (new_rho - self.rho[l])
                self.rho[l] = max(0.0, min(1.0, self.rho[l]))
            # Update weight importance ratio nu_l
            if len(weight_vars) > l:
                var_w = weight_vars[l]
                new_nu = max(0.0, min(1.0, self.nu[l] * (var_w / self.tau_w)))
                self.nu[l] += self.alpha * (new_nu - self.nu[l])
                self.nu[l] = max(0.0, min(1.0, self.nu[l]))

    def sample_indices(self, importance_scores: torch.Tensor, ratio: float) -> torch.Tensor:
        """
        Sample data point indices based on importance scores.
        Args:
            importance_scores (Tensor): 1D tensor with importance scores per data point (size N).
            ratio (float): Current importance ratio for data points (0 < ratio <=1).

        Returns:
            mask (Tensor): Boolean tensor of shape (N,), True for sampled points.
        """
        N = importance_scores.shape[0]
        # Convert importance scores to probabilities proportional to importance_scores
        # Ensure sum of probabilities equals ratio * N
        # To avoid small probabilities being zero, add epsilon
        epsilon = 1e-8
        importance_scores = importance_scores + epsilon
        total = importance_scores.sum()
        # Scale probabilities so that sum equals ratio * N
        probs = importance_scores * (ratio * N / total)
        probs = torch.clamp(probs, max=1.0)  # Probabilities capped at 1

        # Bernoulli sampling for each data point
        sampled_mask = torch.bernoulli(probs)
        # Make sure the expected number of samples is close to ratio * N
        return sampled_mask.bool()

    def sample_tokens(self, token_importance_scores: torch.Tensor, ratio: float) -> torch.Tensor:
        """
        Sample tokens within sequences based on importance scores.
        Args:
            token_importance_scores (Tensor): shape [N, T], importance scores of each token.
            ratio (float): importance ratio for tokens in [0,1].

        Returns:
            mask (Tensor): Boolean tensor of shape [N, T], True for retained tokens.
        """
        N, T = token_importance_scores.shape
        # Flatten scores to shape [N*T]
        scores_flat = token_importance_scores.reshape(-1) + 1e-8
        total_score = scores_flat.sum()
        # Total number of tokens across batch
        total_tokens = N * T
        # Compute probabilities proportional to importance scores
        probs = scores_flat * (ratio * total_tokens / total_score)
        probs = torch.clamp(probs, max=1.0)

        # Bernoulli sampling for each token
        sampled_mask_flat = torch.bernoulli(probs)
        sampled_mask = sampled_mask_flat.reshape(N, T).bool()
        return sampled_mask

    def get_activation_importance_scores(self, layer_activation_grads: List[torch.Tensor]) -> List[torch.Tensor]:
        """
        Compute importance scores for activation gradients in each layer.
        Currently: based on gradient norm.
        Args:
            layer_activation_grads (List[Tensor]): List of activation gradients per layer.
            Each tensor shape: [N, T, K] or similar.

        Returns:
            List of importance scores (e.g., Frobenius norms per sample per layer).
        """
        importance_scores: List[torch.Tensor] = []
        for grad in layer_activation_grads:
            # Compute per-sample importance: Frobenius norm over K for each sample
            # Assuming shape [N, T, K]
            # Compute norm over last dimension for each [N, T]
            norm_per_sample = torch.norm(grad, dim=-1, p='fro')  # shape [N, T]
            # Sum over tokens to get per-sample importance
            importance = norm_per_sample.sum(dim=1)  # shape [N]
            importance_scores.append(importance)
        return importance_scores

    def get_weight_importance_scores(self, layer_weight_grads: List[torch.Tensor]) -> List[torch.Tensor]:
        """
        Compute importance scores for weight gradients (e.g., leverage scores).
        For simplicity, use Frobenius norms.
        Args:
            layer_weight_grads (List[Tensor]): List of weight gradient tensors per layer.

        Returns:
            List of importance scores (scalar per layer).
        """
        importance_scores: List[torch.Tensor] = []
        for grad in layer_weight_grads:
            importance = torch.norm(grad, p='fro')
            importance_scores.append(importance)
        return importance_scores

    def update_step(self):
        """
        Increment step count and trigger ratios update if needed.
        """
        self.step += 1
        if self.step % self.F == 0:
