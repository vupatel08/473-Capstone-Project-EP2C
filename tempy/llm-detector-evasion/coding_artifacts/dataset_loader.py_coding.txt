## dataset_loader.py
import os
import json
import random
from typing import List, Tuple, Dict, Optional
import numpy as np
from tqdm import tqdm
from datasets import load_dataset
import torch
from transformers import AutoTokenizer, AutoModelForCausalLM
from detector import Detector
import logging

# Set up basic logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class DatasetLoader:
    """
    Handles dataset loading, prompt sampling, response generation using the base model,
    detector scoring, and preference pair creation.
    """
    def __init__(self, config: dict):
        """
        Initialize dataset loader with configuration.
        """
        self.config = config
        self.detectors_config = config.get("detectors", {})
        self.dataset_config = config.get("dataset", {})
        self.model_config = config.get("model", {})
        self.gen_params = {
            "max_response_tokens": self.dataset_config.get("max_response_tokens", 250),
            "temperature": self.dataset_config.get("generation_temperature", 1.0)
        }
        # Initialize deterministic seed for reproducibility
        seed = self.dataset_config.get("seed", 42)
        random.seed(seed)
        np.random.seed(seed)
        torch.manual_seed(seed)

        # Load model and tokenizer
        self.device = self.model_config.get("device", "cuda")
        self.model_name = self.model_config.get("name", "Llama-2-7b")
        self.model = self._load_model()
        self.tokenizer = self._load_tokenizer()

        # Load datasets
        self.raw_prompts = self._load_prompt_dataset()

        # Initialize detectors
        self.detectors = self._initialize_detectors()

    def _load_model(self):
        """
        Load the language model based on configuration.
        """
        model_name = self.model_name
        logger.info(f"Loading model {model_name}")
        model = AutoModelForCausalLM.from_pretrained(model_name)
        model.to(self.device)
        model.eval()
        return model

    def _load_tokenizer(self):
        """
        Load tokenizer for the model.
        """
        tokenizer = AutoTokenizer.from_pretrained(self.model_name)
        # Ensure padding token if needed
        if tokenizer.pad_token is None:
            tokenizer.add_special_tokens({"pad_token": tokenizer.eos_token})
        return tokenizer

    def _load_prompt_dataset(self):
        """
        Load prompts based on specified dataset source.
        Currently supports 'OpenWebText', 'Instruction', 'RedditPrompts', 'ChatGPT'.
        """
        source = self.dataset_config.get("prompt_dataset", "OpenWebText")
        dataset_name_map = {
            "OpenWebText": "openwebtext",
            "Instruction": "sft_data_instruct",  # custom script or preloaded dataset
            "RedditPrompts": "reddit_writing_prompts",
            "ChatGPT": "chatgpt_prompts"  # hypothetical, assume API or local file
        }
        prompts = []
        if source == "OpenWebText":
            # Load using datasets library if available
            try:
                ds = load_dataset("openwebtext")
                # Use 'text' field if present
                prompts = [item['text'] for item in ds['train']]
                logging.info(f"Loaded {len(prompts)} prompts from OpenWebText")
            except Exception:
                # fallback: load from local file or raise
                raise NotImplementedError("OpenWebText dataset loading not implemented. Please prepare the dataset.")
        elif source == "reddit_writing_prompts":
            try:
                ds = load_dataset("reddit_writing_prompts")
                prompts = [item['prompt'] for item in ds['train']]
                logging.info(f"Loaded {len(prompts)} Reddit prompts.")
            except:
                raise NotImplementedError("Reddit Writing Prompts dataset loading not implemented.")
        elif source == "sft_data_instruct":
            # Placeholder: load instruction prompts from local file or dataset
            # For demonstration, generate synthetic prompts or load from a file
            prompts = ["Write a summary of the last news article.", 
                       "Explain the theory of relativity.",
                       "Describe a historical event.",
                       "Generate a creative story about a dragon."]
            logging.info(f"Using default instruction prompts, count: {len(prompts)}")
        elif source == "chatgpt_prompts":
            # Placeholder: load or generate prompts via API or file
            prompts = ["Discuss the benefits of open-source software.",
                       "Write a poem about the ocean.",
                       "Outline a plan for a science project."]
            logging.info(f"Using default ChatGPT-like prompts, count: {len(prompts)}")
        else:
            raise ValueError(f"Unknown prompt dataset source: {source}")
        return prompts

    def _initialize_detectors(self) -> List[Detector]:
        """
        Set up detector instances based on configuration.
        """
        detectors_list = []
        # Handle open-source detectors
        for det_conf in self.detectors_config.get("open_source", []):
            name = det_conf.get("name")
            detector = Detector(name=name, device=self.device)
            detectors_list.append(detector)
        # Handle commercial detectors with API keys if available
        for det_conf in self.detectors_config.get("commercial", []):
            name = det_conf.get("name")
            api_key = det_conf.get("api_key", None)
            detector = Detector(name=name, api_key=api_key, device=self.device)
            detectors_list.append(detector)
        return detectors_list

    def generate_responses(self, prompt: str, num_responses: int = 2) -> List[str]:
        """
        Generate responses (completions) for a prompt.
        """
        responses = []
        # Generate multiple responses using model.generate
        # For simplicity, generate responses sequentially; batch if needed
        inputs = self.tokenizer(prompt, return_tensors='pt').to(self.device)
        prompt_ids = inputs['input_ids']
        attention_mask = inputs['attention_mask']
        for _ in range(num_responses):
            with torch.no_grad():
                output_ids = self.model.generate(
                    input_ids=prompt_ids,
                    attention_mask=attention_mask,
                    max_new_tokens=self.gen_params["max_response_tokens"],
                    temperature=self.gen_params["temperature"],
                    do_sample=True,
                    top_p=0.9,
                    top_k=50,
                    pad_token_id=self.tokenizer.eos_token_id,
                )
            response_ids = output_ids[0][prompt_ids.shape[-1]:]
            response_text = self.tokenizer.decode(response_ids, skip_special_tokens=True)
            responses.append(response_text.strip())
        return responses

    def score_responses(self, prompts: List[str], responses: List[str]) -> Dict[str, List[float]]:
        """
        Score each response with each detector: returns dict[detector_name] -> list of scores
        """
        detector_scores = {det.name: [] for det in self.detectors}
        batch_size = 16  # For efficient batching; adjust depending on memory
        total = len(prompts)
        for start in tqdm(range(0, total, batch_size), desc="Scoring responses"):
            end = min(start + batch_size, total)
            batch_prompts = prompts[start:end]
            batch_responses = responses[start:end]
            # For each detector, score the batch
            for det in self.detectors:
                scores = det.batch_score(list(zip(batch_prompts, batch_responses)))
                detector_scores[det.name].extend(scores)
        return detector_scores

    def create_preference_dataset(self, num_pairs: int) -> List[dict]:
        """
        Generate the preference pairs dataset.
        For each prompt:
            - Generate 2 responses
            - Score responses with detectors
            - For each detector:
                - Determine preference based on scores
                - Store pair with label
        Return list of dicts: each dict contains prompt, response_a, response_b, preferred_response, detector_scores
        """
        preference_pairs = []
        prompts_sample = self._sample_prompts(num_pairs)
        logger.info(f"Creating preferences for {len(prompts_sample)} prompts.")

        for prompt in tqdm(prompts_sample, desc="Generating preference pairs"):
            responses = self.generate_responses(prompt, num_responses=2)
            # Score responses with each detector
            detector_scores = self.score_responses([prompt], responses)
            # For each detector, create a preference pair
            for det_name, scores in detector_scores.items():
                score_a, score_b = scores[0], scores[1]
                if score_a > score_b:
                    preferred_response = responses[0]
                    response_a, response_b = responses[0], responses[1]
                elif score_b > score_a:
                    preferred_response = responses[1]
                    response_a, response_b = responses[0], responses[1]
                else:
                    # If scores are equal, skip or randomly assign preference
                    # Here, we assign response_a as preferred for stability
                    response_a, response_b = responses[0], responses[1]
                    preferred_response = responses[0]
                pair = {
                    "prompt": prompt,
                    "response_a": response_a,
                    "response_b": response_b,
                    "preferred_response": preferred_response,
                    "detector": det_name,
                    "detector_scores": {
                        "score_response_a": scores[0],
                        "score_response_b": scores[1]
                    }
                }
                preference_pairs.append(pair)
        return preference_pairs

    def _sample_prompts(self, num_samples: int) -> List[str]:
        """
        Sample prompts uniformly at random from the loaded dataset prompts.
        """
        if len(self.raw_prompts) <= num_samples:
            return self.raw_prompts
        return random.sample(self.raw_prompts, num_samples)
