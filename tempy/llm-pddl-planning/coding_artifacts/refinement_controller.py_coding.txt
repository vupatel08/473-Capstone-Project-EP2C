import os
import numpy as np
from typing import Tuple, List, Dict, Any
from datasets import DatasetLoader
from environment import Environment
from prompt_manager import PromptManager
from pddl_utils import PDDLUtils
from planner_interface import PlannerInterface
import openai
import copy
import logging

# Setup basic logger
logger = logging.getLogger(__name__)
logger.setLevel(logging.INFO)

# Load configuration
import yaml
with open('config.yaml', 'r', encoding='utf-8') as f:
    cfg = yaml.safe_load(f)

# Extract parameters from config
N_P = cfg['generation'].get('n_p', 5)
N_D = cfg['generation'].get('n_d', 10)
C_MAX = cfg['generation'].get('c_max', 4)
T_MAX = cfg['generation'].get('T_max', 10)
TEMP_INITIAL = cfg['generation'].get('temperature_initial', 0.0)
TEMP_REFINEMENT = cfg['generation'].get('temperature_refinement', 0.7)

SUCCESS_THRESHOLD = cfg['evaluation'].get('success_threshold', 0.5)

# We'll assume the prompt templates have been loaded elsewhere, passed to PromptManager
# For simplification, define dummy templates here (should load from actual templates)
initial_prompt_template = cfg['prompting'].get('initial_prompt_template', '')
refinement_prompt_template = cfg['prompting'].get('refinement_prompt_template', '')
problem_prompt_template = cfg['prompting'].get('problem_prompt_template', '')

# Placeholders for the main classes (imports assumed)
# class DatasetLoader, class Environment, class PromptManager, class PDDLUtils, class PlannerInterface

class RefinementController:
    def __init__(
        self,
        environment: Environment,
        prompt_manager: PromptManager,
        pddl_utils: PDDLUtils,
        planner: PlannerInterface,
        env_name: str,
        ground_truth_domain: str,
        ground_truth_problems: List[str],
        nl_descriptions: Dict[str, str],
        environment_objects: List[str]
    ):
        self.env = environment
        self.prompt_manager = prompt_manager
        self.pddl_utils = pddl_utils
        self.planner = planner
        self.env_name = env_name
        self.gt_domain = ground_truth_domain
        self.gt_problems = ground_truth_problems
        self.nl_descs = nl_descriptions
        self.env_objects = environment_objects

        # Initialize current best PDDL files as None (will generate)
        self.current_domain = None
        self.current_problem_template = None  # For initial problem template
        self.best_domain_str = None
        self.best_problem_str = None

    def initialize(self) -> Tuple[str, str]:
        """
        Generate initial domain and problem PDDL using GPT (temperature=0).
        """
        # Generate initial domain
        prompt_d = self.prompt_manager.generate_initial_prompt(
            self.nl_descs.get('domain', ''),
            self.nl_descs.get('problems', [])
        )
        domain_response = self.generate_response(prompt_d, temperature=TEMP_INITIAL)
        # Process and validate domain
        domain_code = domain_response.strip()
        if not self.pddl_utils.validate_pddl_syntax(domain_code):
            logger.warning("Initial domain PDDL syntax invalid, attempting regeneration.")
            # Regenerate with GPT
            domain_response = self.generate_response(prompt_d, temperature=TEMP_INITIAL)
            domain_code = domain_response.strip()
        self.pddl_utils.write_pddl_files(domain_code, "", "temp")  # Empty problem for now
        self.current_domain = domain_code
        # Generate initial problem
        prompt_p = self.prompt_manager.generate_problem_prompt(
            pddl_template="",  # Not used here, placeholder
            env_objects=self.env_objects,
            nl_task=self.nl_descs.get('problems', '')
        )
        problem_response = self.generate_response(prompt_p, temperature=TEMP_INITIAL)
        problem_code = problem_response.strip()
        if not self.pddl_utils.validate_pddl_syntax(problem_code):
            logger.warning("Initial problem PDDL syntax invalid, attempting regeneration.")
            problem_response = self.generate_response(prompt_p, temperature=TEMP_INITIAL)
            problem_code = problem_response.strip()
        # Save files
        self.pddl_utils.write_pddl_files(domain_code, problem_code, "temp")
        self.best_domain_str = domain_code
        self.best_problem_str = problem_code
        return domain_code, problem_code

    def generate_response(self, prompt: str, temperature: float = 0.0) -> str:
        """
        Call GPT-4 API with given prompt.
        """
        # Placeholder: replace with actual API call
        response = openai.ChatCompletion.create(
            model='gpt-4',
            messages=[{"role": "system", "content": prompt}],
            temperature=temperature,
            max_tokens=2000,
        )
        # Parse response depending on API format, assume 'content'
        return response['choices'][0]['message']['content']

    def generate_candidates(self, domain_str: str, problem_str: str) -> Tuple[List[str], List[str]]:
        """
        Generate multiple candidate domain and problem files via GPT.
        """
        # Prepare prompt for candidate generation
        prompt_d = self.prompt_manager.generate_initial_prompt(
            self.nl_descs.get('domain', ''),
            self.nl_descs.get('problems', [])
        )
        # Generate n_d domain candidates with temperature=TEMP_REFINEMENT
        domain_candidates = []
        for _ in range(N_D):
            resp = self.generate_response(prompt_d, temperature=TEMP_REFINEMENT)
            code = resp.strip()
            if self.pddl_utils.validate_pddl_syntax(code):
                domain_candidates.append(code)
        # Generate n_p problem candidates
        problem_candidates = []
        for _ in range(N_P):
            prompt_p = self.prompt_manager.generate_problem_prompt(
                pddl_template="", env_objects=self.env_objects, nl_task=self.nl_descs.get('problems', '')
            )
            resp = self.generate_response(prompt_p, temperature=TEMP_REFINEMENT)
            code = resp.strip()
            if self.pddl_utils.validate_pddl_syntax(code):
                problem_candidates.append(code)
        return domain_candidates, problem_candidates

    def sample_action_sequences(
        self,
        domain_str: str,
        problem_str: str,
        T_max: int
    ) -> List[List[str]]:
        """
        Sample multiple random action sequences from the environment, checking executability.
        Returns list of executable action sequences.
        """
        action_seqs = []
        for _ in range(N_P):  # Number of sequences to sample
            seq = []
            # Initialize environment state
            env_state = copy.deepcopy(self.env.get_current_state())
            for t in range(T_max):
                possible_actions = self.env.get_action_interface()  # List or dict
                actions_list = self.get_possible_actions(possible_actions)
                # Choose randomly among executable actions
                executable_actions = []
                for act in actions_list:
                    if self.env.check_executability([act]):
                        executable_actions.append(act)
                if not executable_actions:
                    break
                chosen_action = np.random.choice(executable_actions)
                success = self.env.execute_action(chosen_action)
                seq.append(chosen_action)
                if not success:
                    # Execution failed, break
                    break
            # Check if entire sequence is executable
            if self.env.check_executability(seq):
                action_seqs.append(seq)
        return action_seqs

    def get_possible_actions(self, action_interface: Dict) -> List[str]:
        """
        Generate list of possible actions from environment's action interface.
        Placeholder: assume environment provides list, in real case, generate accordingly.
        """
        # For placeholder, assume a fixed set of actions
        actions = []
        for act_name in action_interface.get('actions', []):
            actions.append(act_name)
        return actions

    def compute_ew_score(
        self,
        domain_str: str,
        problem_str: str,
        T_max: int
    ) -> float:
        """
        Sample multiple action sequences, compute fraction that are executable - EW score.
        """
        sequences = self.sample_action_sequences(domain_str, problem_str, T_max)
        if not sequences:
            return 0.0
        executable_count = 0
        for q in sequences:
            if self.env.check_executability(q):
                executable_count += 1
        ew_score = executable_count / len(sequences)
        return ew_score

    def run_plan_and_validate(self, domain_str: str, problem_str: str) -> bool:
        """
        Run classical planner, then validate plan with environment (`V_{d,p}`).
        Returns True if plan solves environment.
        """
        # Save PDDL files temporarily
        self.pddl_utils.write_pddl_files(domain_str, problem_str, "temp")
        domain_path = os.path.abspath("temp/domain.pddl")
        problem_path = os.path.abspath("temp/problem.pddl")
        plan_actions = self.planner.run_planner(domain_path, problem_path)
        if plan_actions is None:
            return False
        # Validate plan against environment
        is_valid = self.env.validate_plan(plan_actions)
        return is_valid

    def refine(
        self,
        prev_domain_str: str,
        prev_problem_str: str,
        env_feedback: str,
        plan_success: bool
    ) -> Tuple[str, str, str]:
        """
        Generate refinement PDDL files via GPT given feedback.
        """
        prompt = self.prompt_manager.generate_refinement_prompt(
            prev_domain_str, prev_problem_str, env_feedback, env_feedback
        )
        resp = self.generate_response(prompt, temperature=TEMP_REFINEMENT)
        new_domain_str = resp.strip()
        if not self.pddl_utils.validate_pddl_syntax(new_domain_str):
            logger.warning("Refined domain PDDL invalid, fallback to previous.")
            new_domain_str = prev_domain_str
        # Generate new problem based on the same NL task
        prompt_p = self.prompt_manager.generate_problem_prompt(
            pddl_template="", env_objects=self.env_objects, nl_task=self.nl_descs.get('problems', '')
        )
        resp_p = self.generate_response(prompt_p, temperature=TEMP_REFINEMENT)
        new_problem_str = resp_p.strip()
        if not self.pddl_utils.validate_pddl_syntax(new_problem_str):
            logger.warning("Refined problem PDDL invalid, fallback to previous.")
            new_problem_str = prev_problem_str
        return new_domain_str, new_problem_str, env_feedback

    def execute_refinement(self, nl_descriptions: Dict[str, str], nl_task: str, max_iterations: int) -> Dict[str, Any]:
        """
        Main iterative refinement loop based on Algorithm 1.
        """
        # Step 1: Initialize with initial PDDL
        domain_str, problem_str = self.initialize()
        current_domain_str = domain_str
        current_problem_str = problem_str

        metrics = {
            'solve_rates': [],
            'ew_scores': [],
            'iterations': 0,
            'best_domain': None,
            'best_problem': None
        }

        for cycle in range(max_iterations):
            logger.info(f"Refinement iteration {cycle+1}/{max_iterations}")
            # Generate candidate domain/problem files
            domain_candidates, problem_candidates = self.generate_candidates(current_domain_str, current_problem_str)
            # For each candidate, sample action sequences and compute EW
            candidate_scores = []
            for d_candidate in domain_candidates:
                for p_candidate in problem_candidates:
                    ew_score = self.compute_ew_score(d_candidate, p_candidate, T_MAX)
                    candidate_scores.append({
                        'domain': d_candidate,
                        'problem': p_candidate,
                        'ew_score': ew_score
                    })
            if not candidate_scores:
                logger.warning("No valid candidates generated, stopping refinement.")
                break
            # Select candidate with highest EW score
            candidate_scores.sort(key=lambda x: x['ew_score'], reverse=True)
            top_candidate = candidate_scores[0]
            best_domain = top_candidate['domain']
            best_problem = top_candidate['problem']
            best_ew = top_candidate['ew_score']

            # Run plan and validate in environment
            plan_solves = self.run_plan_and_validate(best_domain, best_problem)
            # Record metrics
            metrics['ew_scores'].append(best_ew)
            solve_success = int(plan_solves)
            metrics['solve_rates'].append(solve_success)
            # Check if better than previous
            if len(metrics['solve_rates']) == 1 or solve_success >= max(metrics['solve_rates']):
                metrics['best_domain'] = best_domain
                metrics['best_problem'] = best_problem

            # Evaluate if success threshold met
            if solve_success >= (len(metrics['solve_rates']) * 1.0) * SUCCESS_THRESHOLD:
                logger.info("Success threshold achieved, stopping refinement.")
                break

            # Prepare environment feedback (simulate or get from environment)
            env_feedback = self.get_environment_feedback()
            # Re-apply environment interaction with selected candidate
            # Alternatively, perform environment sampling to get real feedback
            # For simplicity, re-use previous environmental feedback
            # Perform a refinement step
            new_domain_str, new_problem_str, env_feedback = self.refine(
                best_domain, best_problem, env_feedback, plan_solves
            )
            current_domain_str = new_domain_str
            current_problem_str = new_problem_str
            self.best_domain_str = new_domain_str
            self.best_problem_str = new_problem_str
            metrics['iterations'] += 1

        # Return metrics and best PDDL files
        return {
            'final_domain': self.best_domain_str,
            'final_problem': self.best_problem_str,
            'solve_rates': metrics['solve_rates'],
            'ew_scores': metrics['ew_scores'],
            'iterations': metrics['iterations']
        }

    def get_environment_feedback(self) -> str:
        """
        Obtain environment feedback for the current candidate PDDL.
        Placeholder: could be performed by sampling execution failures or success.
        """
        # For demo, return empty string
        return "Environment feedback placeholder."
