import concurrent.futures
from typing import Dict, List

class Sampler:
    def __init__(self, model, n: int = 20, m: int = 20, temperature: float = 1.0):
        """
        Initialize the Sampler with a model, number of trajectories, max length, and temperature.
        Args:
            model (ModelWrapper): Pre-loaded model wrapper instance.
            n (int): Number of trajectories to sample per prompt.
            m (int): Max tokens per trajectory.
            temperature (float): Sampling temperature.
        """
        self.model = model
        self.n = n
        self.m = m
        self.temperature = temperature

    def sample_trajectories(self, prompt: str) -> List[str]:
        """
        Generate trajectories (continuations) for a single prompt.
        Args:
            prompt (str): Input prompt string.
        Returns:
            List[str]: List of sampled trajectories (sequences).
        """
        # Use ModelWrapper's generate_continuations
        trajectories = self.model.generate_continuations(
            prompt, self.n, self.m, self.temperature
        )
        return trajectories

    def parallel_sample(self, prompts: List[str]) -> Dict[str, List[str]]:
        """
        Batch sampling for multiple prompts in parallel.
        Args:
            prompts (List[str]): List of prompt strings.
        Returns:
            Dict[str, List[str]]: Mapping from prompt to list of trajectories.
        """
        results: Dict[str, List[str]] = {}
        # Use ThreadPoolExecutor for parallelization (can switch to ProcessPoolExecutor if needed)
        with concurrent.futures.ThreadPoolExecutor() as executor:
            # Launch sampling jobs for all prompts
            futures = {
                prompt: executor.submit(self.sample_trajectories, prompt) for prompt in prompts
            }
            for prompt, future in futures.items():
                try:
                    trajectories = future.result()
                    results[prompt] = trajectories
                except Exception as e:
                    # In case of error, fallback to empty list and log error
                    results[prompt] = []
        return results
