```python
class LikelihoodCalculator:
    def __init__(self, model_wrapper, cache=None):
        self.model = model_wrapper
        self.cache = cache if cache is not None else {}

    def compute_log_prob(self, sequence: str, prompt: str) -> float:
        key = self._create_cache_key(sequence, prompt)
        if key in self.cache:
            return self.cache[key]
        log_prob = self._calculate_sequence_log_prob(sequence, prompt)
        self.cache[key] = log_prob
        return log_prob

    def _calculate_sequence_log_prob(self, sequence: str, prompt: str) -> float:
        # Tokenize prompt and sequence
        prompt_tokens = self.model.tokenize(prompt)
        sequence_tokens = self.model.tokenize(sequence)
        # Generate total token sequence
        full_tokens = prompt_tokens + sequence_tokens
        # Get token probabilities for full sequence
        probs = self.model.get_token_probs(full_tokens[:-1], full_tokens[1:])
        # Compute sum of log probs for the sequence
        log_probs = torch.log(probs)
        total_log_prob = log_probs.sum().item()
        return total_log_prob

    def _create_cache_key(self, sequence, prompt):
        # Concatenate prompt and sequence tokens as key
        return tuple(self.model.tokenize(prompt + sequence))
```
