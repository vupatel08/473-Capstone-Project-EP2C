BATCH_SIZE = config.get('evaluation', {}).get('evaluation_batch_size', 128)

REAL_STATS_PATH = config.get('evaluation', {}).get('real_stats_path', None)

# Initialize Inception v3 model for feature extraction
# We use the 'pool3' features (with default last pooling).
# In torchvision, feature extraction can be done by removing the classifier head.
class InceptionFeatureExtractor(nn.Module):
    def __init__(self):
        super().__init__()
        self.model = inception_v3(pretrained=True, transform_input=False)
        self.model.eval()
        # Remove final classifier
        self.features = nn.Sequential(*list(self.model.children())[:-1])  # before fc
        self.model.to(device)
        # Disable gradients
        for param in self.model.parameters():
            param.requires_grad = False

    def extract_features(self, images: torch.Tensor) -> torch.Tensor:
        # images: [batch, 3, 299, 299], normalized
        with torch.no_grad():
            feats = self.features(images)
            # feats shape: [batch, 2048, 8, 8]
            # Global average pooling
            feats = F.adaptive_avg_pool2d(feats, (1,1)).squeeze(-1).squeeze(-1)
            return feats.cpu().numpy()

# Initialize
inception = InceptionFeatureExtractor()

# Normalization transforms for images
transform = Compose([
    Resize((299, 299)),
    ToTensor(),
    Normalize(mean=[0.485, 0.456, 0.406],
              std=[0.229, 0.224, 0.225]),
])

def preprocess_images(images: List[Image.Image]) -> torch.Tensor:
    """
    Convert list of PIL images to batch tensor normalized for Inception.
    """
    tensors = [transform(img) for img in images]
    batch = torch.stack(tensors, dim=0).to(device)
    return batch

def get_image_features(images: List[Image.Image]) -> np.ndarray:
    """
    Batch process images to extract features.
    """
    all_feats = []
    for i in tqdm(range(0, len(images), BATCH_SIZE), desc='Extracting features'):
        batch_images = images[i:i+BATCH_SIZE]
        batch_tensor = preprocess_images(batch_images)
        feats = inception.extract_features(batch_tensor)
        all_feats.append(feats)
    return np.concatenate(all_feats, axis=0)

def load_real_stats(stats_path: str):
    """
    Load precomputed real dataset statistics for FID.
    Expected to be a .npz file with 'mean' and 'cov'.
    """
    data = np.load(stats_path)
    return data['mean'], data['cov']

def compute_statistics(feats: np.ndarray):
    """
    Compute mean and covariance of features.
    """
    mu = np.mean(feats, axis=0)
    sigma = np.cov(feats, rowvar=False)
    return mu, sigma

def calculate_fid(mu1, sigma1, mu2, sigma2):
    """
    Compute the Fréchet Inception Distance (FID).
    """
    diff = mu1 - mu2
    covmean, _ = linalg.sqrtm(sigma1 @ sigma2, disp=False)
    # Numerical stability
    if np.iscomplexobj(covmean):
        covmean = covmean.real
    fid = diff.dot(diff) + np.trace(sigma1 + sigma2 - 2 * covmean)
    return fid

def compute_fid(generated_images: List[Image.Image]) -> float:
    """
    Compute FID between generated images and real dataset stats.
    """
    # Extract features from generated images
    gen_feats = get_image_features(generated_images)
    mu_gen, sigma_gen = compute_statistics(gen_feats)

    # Load real stats
    if REAL_STATS_PATH is None:
        raise RuntimeError("Path to real dataset statistics must be specified in config.")
    mu_real, sigma_real = load_real_stats(REAL_STATS_PATH)

    fid_score = calculate_fid(mu_gen, sigma_gen, mu_real, sigma_real)
    return float(fid_score)

def compute_inception_score(generated_images: List[Image.Image], splits: int = 10) -> float:
    """
    Compute Inception Score for generated images.
    """
    all_probs = []
    with torch.no_grad():
        for i in tqdm(range(0, len(generated_images), BATCH_SIZE), desc='Computing IS'):
            batch_images = generated_images[i:i+BATCH_SIZE]
            batch_tensor = preprocess_images(batch_images)
            logits = inception.model(batch_tensor)
            # Apply softmax to get class probabilities
            probs = F.softmax(logits, dim=1)  # [batch, 1000]
            all_probs.append(probs.cpu().numpy())
    all_probs = np.concatenate(all_probs, axis=0)  # [N, 1000]

    # Compute marginal distribution
    split_size = len(all_probs) // splits
    split_probs = np.array_split(all_probs, splits, axis=0)
    scores = []
    for probs in split_probs:
        p_mean = np.mean(probs, axis=0)
        kl_divs = probs * (np.log(probs + 1e-10) - np.log(p_mean + 1e-10))
        kl_div = np.sum(kl_divs, axis=1)
        scores.append(np.exp(np.mean(kl_div)))
    return float(np.mean(scores))

def compute_metrics(generated_images: List[Image.Image]) -> dict:
    """
    Compute all metrics: FID, IS, Precision, Recall
    """
    # Extract features
    feats = get_image_features(generated_images)
