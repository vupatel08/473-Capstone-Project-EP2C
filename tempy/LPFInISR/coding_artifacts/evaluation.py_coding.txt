## evaluation.py
import numpy as np
import torch
import torch.nn.functional as F
from skimage.metrics import structural_similarity as compare_ssim
import lpips
import matplotlib.pyplot as plt
import yaml
from scipy.fft import fft2, fftshift
from scipy.signal import get_window
import os
from typing import Tuple

# Load configuration parameters from config.yaml
with open('config.yaml', 'r') as f:
    cfg = yaml.safe_load(f)

FFT_SIZE: int = cfg['spectral_analysis'].get('fft_size', 256)
WINDOW_FUNC: str = cfg['spectral_analysis'].get('window_function', 'hann')
NORMALIZE_SPECTRA: bool = cfg['spectral_analysis'].get('spectral_normalization', True)

OUTPUT_DIR: str = cfg['utilities'].get('output_dir', './results')
os.makedirs(OUTPUT_DIR, exist_ok=True)

# Initialize LPIPS model
lpips_model = lpips.LPIPS(net='vgg')  # default to VGG, can be changed as needed

def compute_psnr(hr_img: np.ndarray, sr_img: np.ndarray, max_val: float = 1.0) -> float:
    """
    Compute PSNR between high-resolution and super-resolved images.
    Args:
        hr_img (np.ndarray): HR image, shape (H,W,C).
        sr_img (np.ndarray): SR image, shape (H,W,C).
        max_val (float): Maximum pixel value (e.g., 1.0 or 255).
    Returns:
        float: PSNR in dB.
    """
    mse = np.mean((hr_img - sr_img) ** 2)
    if mse == 0:
        return float('inf')
    psnr = 20 * np.log10(max_val / np.sqrt(mse))
    return psnr

def compute_ssim(hr_img: np.ndarray, sr_img: np.ndarray) -> float:
    """
    Compute SSIM between HR and SR images.
    Args:
        hr_img (np.ndarray): HR image, shape (H,W,C).
        sr_img (np.ndarray): SR image, shape (H,W,C).
    Returns:
        float: SSIM index.
    """
    ssim = compare_ssim(hr_img, sr_img, multichannel=True, data_range=1.0)
    return ssim

def compute_lpips(hr_img: np.ndarray, sr_img: np.ndarray) -> float:
    """
    Compute LPIPS distance between HR and SR images.
    Args:
        hr_img (np.ndarray): HR image, shape (H,W,C).
        sr_img (np.ndarray): SR image, shape (H,W,C).
    Returns:
        float: LPIPS distance.
    """
    # Ensure images are 3D tensors with shape (C, H, W)
    hr_tensor = torch.from_numpy(hr_img).permute(2, 0, 1).unsqueeze(0)
    sr_tensor = torch.from_numpy(sr_img).permute(2, 0, 1).unsqueeze(0)
    # Normalize to [0,1] if not already
    with torch.no_grad():
        dist = lpips_model(hr_tensor, sr_tensor)
    return dist.item()

def get_window_2d(size: Tuple[int, int], window_name: str = 'hann') -> np.ndarray:
    """
    Generate a 2D window for spectral analysis.
    Args:
        size (Tuple[int, int]): (height, width) of window.
        window_name (str): Window function name.
    Returns:
        np.ndarray: 2D window.
    """
    h, w = size
    window_x = get_window(window_name, w)
    window_y = get_window(window_name, h)
    window_2d = np.outer(window_y, window_x)
    return window_2d

def compute_fft(image: np.ndarray) -> np.ndarray:
    """
    Compute centered 2D FFT spectrum with windowing and zero-padding.
    Args:
        image (np.ndarray): 2D grayscale or 3D color image.
    Returns:
        np.ndarray: Spectrum (H, W, C) if color, else (H, W)
    """
    # If color, process each channel
    if image.ndim == 3:
        channels = []
        for c in range(image.shape[2]):
            channel = image[:, :, c]
            spectrum = _compute_single_fft(channel)
            channels.append(spectrum)
        return np.stack(channels, axis=-1)
    else:
        return _compute_single_fft(image)

def _compute_single_fft(image: np.ndarray) -> np.ndarray:
    """
    Helper to compute FFT of a single-channel image with windowing.
    """
    h, w = image.shape
    # Zero-pad to FFT_SIZE if smaller
    pad_h = max(FFT_SIZE - h, 0)
    pad_w = max(FFT_SIZE - w, 0)
    pad_top = pad_h // 2
    pad_bottom = pad_h - pad_top
    pad_left = pad_w // 2
    pad_right = pad_w - pad_left
    padded = np.pad(image, ((pad_top, pad_bottom), (pad_left, pad_right)), mode='constant', constant_values=0)
    # Apply window
    window2d = get_window_2d(padded.shape, window_name=WINDOW_FUNC)
    windowed = padded * window2d
    # FFT
    fft_res = fft2(windowed, s=(FFT_SIZE, FFT_SIZE))
    fft_res = fftshift(fft_res)
    return fft_res

def normalize_spectrum(spectrum: np.ndarray) -> np.ndarray:
    """
    Normalize spectrum magnitude to zero-mean, unit-variance.
    Args:
        spectrum (np.ndarray): Complex spectrum.
    Returns:
        np.ndarray: Normalized spectrum (complex).
    """
    mag = np.abs(spectrum)
    mean_val = np.mean(mag)
    std_val = np.std(mag) + 1e-8
    mag_norm = (mag - mean_val) / std_val
    # Clip for stability
    mag_norm = np.clip(mag_norm, -3, 3)
    phase = np.angle(spectrum)
    return mag_norm * np.exp(1j * phase)

def calculate_spectrum(image: np.ndarray) -> np.ndarray:
    """
    Compute normalized magnitude spectrum of the image.
    Args:
        image (np.ndarray): 2D or 3D image.
    Returns:
        np.ndarray: Spectrum magnitude.
    """
    spectrum = compute_fft(image)
    mag = np.abs(spectrum)
    if NORMALIZE_SPECTRA:
        mag = (mag - np.mean(mag)) / (np.std(mag) + 1e-8)
    return mag

def plot_spectrum(spectrum: np.ndarray, title: str = 'Spectrum'):
    """
    Plot the magnitude spectrum.
    """
    plt.figure()
    plt.imshow(np.log10(np.abs(spectrum) + 1e-8), cmap='inferno')
    plt.colorbar()
    plt.title(title)
    plt.tight_layout()
    plt.savefig(os.path.join(OUTPUT_DIR, title.replace(' ', '_') + '.png'))
    plt.close()

def extract_impulse_response(model: torch.nn.Module, device: torch.device) -> np.ndarray:
    """
    Generate impulse response by passing an impulse image.
    """
    size = 11
    impulse_np = np.zeros((size, size, 3), dtype=np.float32)
    center = size // 2
    impulse_np[center, center, :] = 1.0
    input_tensor = torch.from_numpy(impulse_np.transpose(2,0,1)).unsqueeze(0).to(device)
    with torch.no_grad():
        output = model(input_tensor)
    output_np = output.squeeze(0).permute(1,2,0).cpu().numpy()
    return output_np

def spectral_similarity(spectrum_hr: np.ndarray, spectrum_sr: np.ndarray) -> float:
    """
    Compute the spectral distribution similarity (FSDS) in dB.
    """
    # Normalize if required
    if NORMALIZE_SPECTRA:
        spectrum_hr = (spectrum_hr - np.mean(spectrum_hr)) / (np.std(spectrum_hr) + 1e-8)
        spectrum_sr = (spectrum_sr - np.mean(spectrum_sr)) / (np.std(spectrum_sr) + 1e-8)
    diff = np.abs(spectrum_hr - spectrum_sr)
    sum_diff = np.sum(diff)
    epsilon = 1e-8
    fsds_value = 10.0 * np.log10(sum_diff + epsilon)
    return fsds_value

def calculate_fsds(hr_img: np.ndarray, sr_img: np.ndarray) -> float:
    """
    Calculate FSDS metric between HR and SR images.
    """
    spectrum_hr = compute_fft(hr_img)
    spectrum_sr = compute_fft(sr_img)
    mag_hr = np.abs(spectrum_hr)
    mag_sr = np.abs(spectrum_sr)
    if NORMALIZE_SPECTRA:
        mag_hr = (mag_hr - np.mean(mag_hr)) / (np.std(mag_hr)+1e-8)
        mag_sr = (mag_sr - np.mean(mag_sr)) / (np.std(mag_sr)+1e-8)
    fsds_score = spectral_similarity(mag_hr, mag_sr)
    return fsds_score

def simulate_lowpass_filter(cutoff_freq: float, size: Tuple[int, int], window_name: str = 'hann') -> np.ndarray:
    """
    Create a 2D ideal low-pass filter mask in frequency domain.
    Args:
        cutoff_freq (float): cutoff frequency in normalized radians (0 to pi).
        size (Tuple[int,int]): spatial size (H,W).
        window_name (str): window name.
    Returns:
        np.ndarray: 2D filter mask.
    """
    h, w = size
    freq_x = np.fft.fftfreq(w)
    freq_y = np.fft.fftfreq(h)
    FX, FY = np.meshgrid(freq_x, freq_y)
    radius = np.sqrt(FX**2 + FY**2)
    # Convert cutoff to normalized frequency (0 to 0.5 cycles/pixel)
    norm_cutoff = cutoff_freq / np.pi
    mask = np.zeros_like(radius)
    mask[radius <= norm_cutoff] = 1.0
    if window_name:
        window_x = get_window(window_name, w)
        window_y = get_window(window_name, h)
        window2d = np.outer(window_y, window_x)
        mask *= window2d
    return mask

def plot_spectra_comparison(spectrum1, spectrum2, title1='Spectrum 1', title2='Spectrum 2'):
    """
    Plot two spectra in log scale side by side.
    """
    plt.figure(figsize=(10,4))
    plt.subplot(1,2,1)
    plt.imshow(np.log10(np.abs(spectrum1)+1e-8), cmap='inferno')
    plt.title(title1)
    plt.colorbar()

    plt.subplot(1,2,2)
    plt.imshow(np.log10(np.abs(spectrum2)+1e-8), cmap='inferno')
    plt.title(title2)
    plt.colorbar()

    plt.tight_layout()
    plt.savefig(os.path.join(OUTPUT_DIR, title1 + '_vs_' + title2 + '.png'))
    plt.close()
