import os
import numpy as np
import torch
import matplotlib.pyplot as plt
import yaml
from dataset_loader import Dataset
from model import get_model
from signal_analysis import (
    extract_impulse_response,
    decompose_response,
    compute_fft,

from utils import (
    set_seed,
    create_output_dir,
    generate_impulse_image,
    plot_spectrum,
)

def main():
    # Load configuration
    with open('config.yaml', 'r') as f:
        config = yaml.safe_load(f)

    # Set seed for reproducibility
    seed = config['utilities'].get('random_seed', 42)
    set_seed(seed)

    output_dir = config['utilities'].get('output_dir', './results')
    create_output_dir(output_dir)

    # Device setup
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')

    # 1. Load Dataset
    dataset_params = config['dataset']
    dataset_path = dataset_params.get('name', './DIV2K')
    split = dataset_params.get('split', 'validation')
    image_size = dataset_params.get('image_size', 128)
    normalize = dataset_params.get('normalize', True)

    dataset = Dataset(dataset_path=dataset_path, split=split, 
                      image_size=image_size, normalize=normalize, seed=seed)

    # For spectral analysis, generate impulse image (11x11 center pixel = 1)
    impulse_img = generate_impulse_image(size=11)

    # 2. Load Model
    model_arch = config['model'].get('architecture', 'EDSR')
    pretrained = config['model'].get('pretrained', False)
    checkpoint_path = config['model'].get('checkpoint_path', '')
    model_cfg = {
        'pretrained': pretrained,
        'checkpoint_path': checkpoint_path
    }
    model = get_model(model_arch, model_cfg).to(device)
    model.eval()

    # 3. Obtain impulse response of the network
    impulse_resp = extract_impulse_response(model, device)

    # Save or plot impulse response for reference
    plt.imshow(impulse_resp.squeeze(), cmap='gray')
    plt.title('Impulse Response H(δ)')
    plt.colorbar()
    plt.savefig(os.path.join(output_dir, 'impulse_response.png'))
    plt.close()

    # 4. Spectral analysis of impulse response (sinc phenomenon)
    fft_resp = compute_fft(impulse_resp.squeeze())
    plot_spectrum(fft_resp, title='FFT of Impulse Response')
    # Check if sinc-like; optional: quantify similarity (e.g., correlation)

    # Feed impulse image through network to get N(I)
    with torch.no_grad():
        N_I_tensor = torch.from_numpy(impulse_img.transpose(2,0,1)).unsqueeze(0).to(device)
        N_I = model(N_I_tensor).squeeze(0).permute(1,2,0).cpu().numpy()
    # Extract H(δ) already obtained as impulse_resp
    H_delta = impulse_resp

    # Decompose into H(I) and G(I)
    H_I, G_I = decompose_response(N_I, H_delta)

    # Save or visualize G(I)
    plt.imshow(np.abs(G_I).max(axis=2), cmap='inferno')
    plt.title('Non-linear component G(I)')
    plt.colorbar()
    plt.savefig(os.path.join(output_dir, 'nonlinear_response.png'))
    plt.close()

    # 6. Spectral analysis of responses
    spectra_H = compute_fft(H_delta.squeeze())
    spectra_N = compute_fft(N_I)
    spectra_G = compute_fft(G_I)

    # Plot amplitude & phase spectra for H(δ)
    plot_spectra_comparison(np.abs(spectra_H), np.abs(spectra_N),
                            title1='H(δ) Magnitude', title2='N(I) Magnitude')
    plot_spectra_comparison(np.angle(spectra_H), np.angle(spectra_N),
                            title1='H(δ) Phase', title2='N(I) Phase')

    # Verify sinc shape for H(δ) (optional similarity metric)
    # Here, visual check suffices, or compute correlation with true sinc

    # For a range of cutoff frequencies, simulate filtering and evaluate performance
    cutoff_freqs = np.linspace(40, 50, num=11)  # as in experiment
    psnr_list = []
    ssim_list = []
    fsds_list = []

    # Generate example low-resolution image
    # Using a real image from dataset for evaluation
    test_img = dataset[0]
    hr_img = test_img
    # For simplicity, generate a synthetic low-res version with bicubic
    # Normally, use the real LR image if available
    # Here, create a downsampled version
    import cv2
    lr_img = cv2.resize(hr_img, dsize=(hr_img.shape[1]//2, hr_img.shape[0]//2),
                        interpolation=cv2.INTER_CUBIC)
    # Upsample with zero-order hold (nearest) or other methods for baseline
    for cutoff in cutoff_freqs:
        # Generate lowpass filter mask
        filter_mask = utils.generate_lowpass_filter(cutoff, size=hr_img.shape[:2],
                                                    window_func=config['spectral_analysis'].get('window_function', 'hann'))

        # Apply lowpass filter to HR image (simulate super-resolution)
        # via FFT & inverse FFT
        spectra_hr = compute_fft(hr_img)
        spectra_hr_filtered = spectra_hr * filter_mask
        # inverse FFT
        hr_fft_shifted = np.fft.ifftshift(spectra_hr_filtered)
        sr_img_fft = np.fft.ifft2(hr_fft_shifted).real
        # Since spectrum is symmetric, get real part

        # Evaluate metrics
        psnr_value = compute_psnr(hr_img, sr_img_fft)
        ssim_value = compute_ssim(hr_img, sr_img_fft)

        spect_sr = compute_fft(sr_img_fft)
        # For FSDS, compute between hr_img & sr_img_fft
        fsds_value = compute_fsds(hr_img, sr_img_fft)

        psnr_list.append(psnr_value)
        ssim_list.append(ssim_value)
        fsds_list.append(fsds_value)

    # Plot PSNR and SSIM vs. cutoff frequency
    plt.figure()
    plt.plot(cutoff_freqs, psnr_list, label='PSNR')
    plt.plot(cutoff_freqs, ssim_list, label='SSIM')
    plt.xlabel('Cutoff Frequency (ω)')
    plt.ylabel('Metric Value')
    plt.legend()

    # For each model, extract impulse response

    plt.title('Spectrum of G(I)')
    plt.colorbar()
    plt.savefig(os.path.join(output_dir, 'G_response_spectrum.png'))
    plt.close()

    # 10. Compute FSDS between HR and SR images at multiple scales
    # For example, for the model's output SR images:
    # Here, assuming you have SR images from the model
    # For demonstration, simulate SR images via simple interpolations

    fsds_score_bicubic = evaluate_fsds(hr_img, sr_bicubic)
